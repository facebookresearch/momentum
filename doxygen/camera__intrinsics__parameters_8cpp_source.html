22// Find the model parameter index for a given full parameter name, or -1 if not present.
23Eigen::Index findParamIndex(const ParameterTransform& paramTransform, const std::string& fullName) {
24 for (size_t i = 0; i < paramTransform.name.size(); ++i) {
25 if (paramTransform.name[i] == fullName) {
26 return static_cast<Eigen::Index>(i);
27 }
28 }
29 return -1;

34std::tuple<ParameterTransform, ParameterLimits> addCameraIntrinsicsParameters(
35 ParameterTransform paramTransform,
36 ParameterLimits paramLimits,
37 const IntrinsicsModel& intrinsicsModel,
38 bool tieFocalLength) {
39 const std::string& effectiveName = intrinsicsModel.name();
40 MT_CHECK(
41 !effectiveName.empty(),
42 "Camera name must not be empty; call setName() on the intrinsics model first.");
43
44 const std::string prefix = makePrefix(effectiveName);
45 const std::string paramSetName = std::string(kIntrinsicsParamPrefix) + effectiveName;
46
47 // Strip existing params for same camera name (idempotency)
48 ParameterSet existingParams = getCameraIntrinsicsParameterSet(paramTransform, effectiveName);
49 if (existingParams.any()) {
50 std::tie(paramTransform, paramLimits) =
51 subsetParameterTransform(paramTransform, paramLimits, ~existingParams);
52 }
53
54 // Get parameter names from the intrinsics model
55 const auto paramNames = intrinsicsModel.getParameterNames();
56
57 // Build the parameter set for this camera
58 ParameterSet paramSet;
59
60 // When tieFocalLength is true, replace separate "fx"/"fy" with a single "f"
61 std::vector<std::string> addedNames;
62 bool fAdded = false;
63 for (const auto& paramName : paramNames) {
64 if (tieFocalLength && (paramName == "fx" || paramName == "fy")) {
65 if (!fAdded) {
66 addedNames.emplace_back(kTiedFocalLengthSuffix);
67 fAdded = true;
68 }
69 // skip fy (already covered by "f")
70 } else {
71 addedNames.push_back(paramName);
72 }
73 }
74
75 // Append parameters
76 paramTransform.name.reserve(paramTransform.name.size() + addedNames.size());
77 for (const auto& name : addedNames) {
78 paramSet.set(paramTransform.name.size());
79 paramTransform.name.push_back(prefix + name);
80 }
81
82 // Store the named parameter set
83 paramTransform.parameterSets[paramSetName] = paramSet;
84
85 // Resize the transform matrix with zero columns for the new parameters
86 paramTransform.transform.conservativeResize(
87 paramTransform.transform.rows(), paramTransform.name.size());
88 paramTransform.transform.makeCompressed();
89
90 return {paramTransform, paramLimits};
91}

93Eigen::VectorXf extractCameraIntrinsics(
94 const ParameterTransform& paramTransform,
95 const ModelParameters& modelParams,
96 const IntrinsicsModel& intrinsicsModel) {
97 // Start with the model's current values as defaults for any missing params
98 Eigen::VectorXf result = intrinsicsModel.getIntrinsicParameters();
99 const std::string prefix = makePrefix(intrinsicsModel.name());
100 const auto paramNames = intrinsicsModel.getParameterNames();
101 const Eigen::Index tiedFIdx = findParamIndex(paramTransform, prefix + kTiedFocalLengthSuffix);
102
103 for (size_t i = 0; i < paramNames.size(); ++i) {
104 auto paramIdx = findParamIndex(paramTransform, prefix + paramNames[i]);
105 if (paramIdx < 0 && tiedFIdx >= 0 && (paramNames[i] == "fx" || paramNames[i] == "fy")) {
106 paramIdx = tiedFIdx;
107 }
108 if (paramIdx >= 0 && paramIdx < modelParams.size()) {
109 result(static_cast<Eigen::Index>(i)) = modelParams(paramIdx);
110 }
111 }
112 return result;
113}

115void setCameraIntrinsics(
116 const ParameterTransform& paramTransform,
117 const IntrinsicsModel& intrinsicsModel,
118 ModelParameters& modelParams) {
119 const std::string prefix = makePrefix(intrinsicsModel.name());
120 const auto paramNames = intrinsicsModel.getParameterNames();
121 const Eigen::VectorXf intrinsicValues = intrinsicsModel.getIntrinsicParameters();
122 const Eigen::Index tiedFIdx = findParamIndex(paramTransform, prefix + kTiedFocalLengthSuffix);
123
124 for (size_t i = 0; i < paramNames.size(); ++i) {
125 auto paramIdx = findParamIndex(paramTransform, prefix + paramNames[i]);
126 if (paramIdx < 0 && tiedFIdx >= 0 && (paramNames[i] == "fx" || paramNames[i] == "fy")) {
127 paramIdx = tiedFIdx;
128 }
129 if (paramIdx >= 0 && paramIdx < modelParams.size()) {
130 modelParams(paramIdx) = intrinsicValues(static_cast<Eigen::Index>(i));
131 }
132 }

159template <typename T>
160CameraIntrinsicsMapping<T>::CameraIntrinsicsMapping(
161 const ParameterTransform& paramTransform,
162 const IntrinsicsModelT<T>& intrinsicsModel)
163 : mutableIntrinsics(intrinsicsModel.clone()) {
164 const std::string prefix = makePrefix(intrinsicsModel.name());
165 const auto paramNames = intrinsicsModel.getParameterNames();
166 modelParamIndices.resize(paramNames.size());
167
168 // Check for tied focal length: a single "f" parameter that controls both fx and fy.
169 const Eigen::Index tiedFIdx = findParamIndex(paramTransform, prefix + kTiedFocalLengthSuffix);
170
171 for (size_t i = 0; i < paramNames.size(); ++i) {
172 modelParamIndices[i] = findParamIndex(paramTransform, prefix + paramNames[i]);
173 // Fall back to tied "f" for fx/fy if separate params not found
174 if (modelParamIndices[i] < 0 && tiedFIdx >= 0 &&
175 (paramNames[i] == "fx" || paramNames[i] == "fy")) {
176 modelParamIndices[i] = tiedFIdx;
177 }
178 }
179}

191template <typename T>
192const IntrinsicsModelT<T>& CameraIntrinsicsMapping<T>::updateIntrinsics(
193 const ModelParametersT<T>& modelParams) {
194 Eigen::VectorX<T> params = mutableIntrinsics->getIntrinsicParameters();
195 for (size_t i = 0; i < modelParamIndices.size(); ++i) {
196 const auto idx = modelParamIndices[i];
197 if (idx >= 0 && idx < modelParams.size()) {
198 params(static_cast<Eigen::Index>(i)) = modelParams(idx);
199 }
200 }
201 mutableIntrinsics->setIntrinsicParameters(params);
202 return *mutableIntrinsics;
203}

205template <typename T>
206void CameraIntrinsicsMapping<T>::addGradient(
207 const Eigen::Matrix<T, 3, Eigen::Dynamic>& J_intrinsics,
208 const Eigen::Vector2<T>& residual,
209 T weight,
210 Eigen::Ref<Eigen::VectorX<T>> gradient) const {
211 for (size_t i = 0; i < modelParamIndices.size(); ++i) {
212 const auto idx = modelParamIndices[i];
213 if (idx >= 0) {
214 const T gradVal = J_intrinsics.col(i).template head<2>().dot(residual);
215 gradient[idx] += weight * gradVal;
216 }
217 }
218}

220template <typename T>
221void CameraIntrinsicsMapping<T>::addJacobian(
222 const Eigen::Matrix<T, 3, Eigen::Dynamic>& J_intrinsics,
223 T weight,
224 Eigen::Index rowOffset,
225 Eigen::Ref<Eigen::MatrixX<T>> jacobian) const {
226 for (size_t i = 0; i < modelParamIndices.size(); ++i) {
227 const auto idx = modelParamIndices[i];
228 if (idx >= 0) {
229 jacobian.template block<2, 1>(rowOffset, idx) +=
230 weight * J_intrinsics.col(i).template head<2>();
231 }
232 }
