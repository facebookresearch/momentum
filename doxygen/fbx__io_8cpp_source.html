123std::vector<MatrixXf> joinTakesOnSharedTimeline(
124 std::span<const std::vector<MatrixXf>> takesPerCharacter) {
125 std::vector<Eigen::Index> takeLengths;
126 for (const auto& takes : takesPerCharacter) {
127 takeLengths.resize(std::max(takeLengths.size(), takes.size()), 0);
128 auto length = takeLengths.begin();
129 for (const MatrixXf& take : takes) {
130 *length = std::max(*length, take.cols());
131 ++length;
132 }
133 }
134 const Eigen::Index numFrames =
135 std::accumulate(takeLengths.begin(), takeLengths.end(), Eigen::Index{0});
136
137 std::vector<MatrixXf> joined;
138 joined.reserve(takesPerCharacter.size());
139 for (const auto& takes : takesPerCharacter) {
140 const auto animated =
141 std::ranges::find_if(takes, [](const MatrixXf& take) { return take.cols() > 0; });
142 if (animated == takes.end()) {
143 joined.emplace_back();
144 continue;
145 }
146 VectorXf pose = VectorXf::Zero(animated->rows());
147 MatrixXf motion(pose.size(), numFrames);
148 Eigen::Index frame = 0;
149 auto take = takes.begin();
150 for (const Eigen::Index length : takeLengths) {
151 const Eigen::Index takeFrames = take != takes.end() ? take->cols() : 0;
152 if (takeFrames > 0) {
153 MT_THROW_IF(
154 take->rows() != pose.size(),
155 "The takes of one character have {} and {} joint parameters.",
156 take->rows(),
157 pose.size());
158 motion.middleCols(frame, takeFrames) = *take;
159 pose = take->col(takeFrames - 1);
160 }
161 motion.middleCols(frame + takeFrames, length - takeFrames).colwise() = pose;
162 frame += length;
163 if (take != takes.end()) {
164 ++take;
165 }
166 }
167 joined.push_back(std::move(motion));
168 }
169 return joined;
170}

189void saveFbxWithJointParams(
190 const filesystem::path& filename,
191 const Character& character,
192 const MatrixXf& jointParams,
193 const double framerate,
194 std::span<const std::vector<Marker>> markerSequence,
195 const FileSaveOptions& options) {
196 FbxBuilder builder;
197 builder.addCharacter(character, options);
198 if (jointParams.cols() != 0 &&
199 jointParams.rows() != character.parameterTransform.numJointParameters()) {
200 MT_LOGE(
201 "Rows of joint values {} do not match joint parameter dimension {} so not saving any motion.",
202 jointParams.rows(),
203 character.parameterTransform.numJointParameters());
204 } else {
205 builder.addMotionWithJointParams(character, framerate, jointParams);
206 }
207 builder.addMarkerSequence(framerate, markerSequence);
208 builder.save(filename);
209}

211void saveFbxWithSkeletonStates(
212 const filesystem::path& filename,
213 const Character& character,
214 std::span<const SkeletonState> skeletonStates,
215 const double framerate,
216 std::span<const std::vector<Marker>> markerSequence,
217 const FileSaveOptions& options) {
218 const size_t nFrames = skeletonStates.size();
219 MatrixXf jointParams(character.parameterTransform.zero().v.size(), nFrames);
220 for (size_t iFrame = 0; iFrame < nFrames; ++iFrame) {
221 jointParams.col(iFrame) =
222 skeletonStateToJointParameters(skeletonStates[iFrame], character.skeleton).v;
223 }
224
225 FbxBuilder builder;
226 builder.addCharacter(character, options);
227 builder.addMotionWithJointParams(character, framerate, jointParams);
228 builder.addMarkerSequence(framerate, markerSequence);
229 builder.save(filename);
230}

239MatrixXf loadFbxBlendShapeWeights(const filesystem::path& filename) {
240 // Initialize FBX SDK
241 auto* manager = ::fbxsdk::FbxManager::Create();
242 auto* ios = ::fbxsdk::FbxIOSettings::Create(manager, IOSROOT);
243 manager->SetIOSettings(ios);
244
245 // RAII-like cleanup for FBX SDK objects
246 ::fbxsdk::FbxScene* scene = nullptr;
247 const auto cleanup = [&]() {
248 if (scene != nullptr) {
249 scene->Destroy();
250 }
251 manager->Destroy();
252 };
253
254 auto* importer = ::fbxsdk::FbxImporter::Create(manager, "");
255 const std::string sFilename = filename.string();
256 if (!importer->Initialize(sFilename.c_str(), -1, manager->GetIOSettings())) {
257 cleanup();
258 return {};
259 }
260
261 scene = ::fbxsdk::FbxScene::Create(manager, "scene");
262 importer->Import(scene);
263 importer->Destroy();
264
265 // Find the animation stack
266 if (scene->GetSrcObjectCount<::fbxsdk::FbxAnimStack>() == 0) {
267 cleanup();
268 return {};
269 }
270 auto* animStack = scene->GetSrcObject<::fbxsdk::FbxAnimStack>(0);
271 if (animStack->GetMemberCount<::fbxsdk::FbxAnimLayer>() == 0) {
272 cleanup();
273 return {};
274 }
275 auto* animLayer = animStack->GetMember<::fbxsdk::FbxAnimLayer>(0);
276
277 // Collect all blend shape channels from all meshes
278 std::vector<::fbxsdk::FbxBlendShapeChannel*> allChannels;
279 for (int iMesh = 0; iMesh < scene->GetSrcObjectCount<::fbxsdk::FbxMesh>(); ++iMesh) {
280 auto* mesh = scene->GetSrcObject<::fbxsdk::FbxMesh>(iMesh);
281 for (int iDeformer = 0; iDeformer < mesh->GetDeformerCount(::fbxsdk::FbxDeformer::eBlendShape);
282 ++iDeformer) {
283 auto* blendShape = static_cast<::fbxsdk::FbxBlendShape*>(
284 mesh->GetDeformer(iDeformer, ::fbxsdk::FbxDeformer::eBlendShape));
285 for (int iChannel = 0; iChannel < blendShape->GetBlendShapeChannelCount(); ++iChannel) {
286 allChannels.push_back(blendShape->GetBlendShapeChannel(iChannel));
287 }
288 }
289 }
290
291 if (allChannels.empty()) {
292 cleanup();
293 return {};
294 }
295
296 // Determine the number of frames from the first animated channel
297 Eigen::Index numFrames = 0;
298 for (auto* channel : allChannels) {
299 auto* curve = channel->DeformPercent.GetCurve(animLayer);
300 if (curve != nullptr) {
301 numFrames = std::max(numFrames, static_cast<Eigen::Index>(curve->KeyGetCount()));
302 }
303 }
304
305 if (numFrames == 0) {
306 cleanup();
307 return {};
308 }
309
310 // Read weights from animation curves
311 const auto numChannels = static_cast<Eigen::Index>(allChannels.size());
312 MatrixXf weights = MatrixXf::Zero(numChannels, numFrames);
313
314 for (Eigen::Index i = 0; i < numChannels; i++) {
315 auto* curve = allChannels[i]->DeformPercent.GetCurve(animLayer);
316 if (curve == nullptr) {
317 continue;
318 }
319
320 const int keyCount = curve->KeyGetCount();
321 for (int k = 0; k < keyCount && k < numFrames; k++) {
322 // FBX DeformPercent is in [0, 100], convert to [0, 1]
323 weights(i, k) = curve->KeyGetValue(k) / 100.0f;
324 }
325 }
326
327 cleanup();
328 return weights;
329}
