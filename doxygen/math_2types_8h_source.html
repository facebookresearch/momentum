465template <typename OtherScalar, template <typename...> class ContainerType, typename ObjectType>
466[[nodiscard]] decltype(auto) cast(const ContainerType<ObjectType>& originalContainer) {
467 using Scalar = typename ObjectType::Scalar;
468
469 if constexpr (std::is_same_v<OtherScalar, Scalar>) {
470 return originalContainer;
471 } else {
472 using CastedType = std::remove_const_t<
473 std::remove_reference_t<decltype(std::declval<ObjectType>().template cast<OtherScalar>())>>;
474
475 ContainerType<CastedType> castedContainer;
476 castedContainer.reserve(originalContainer.size());
477
478 std::transform(
479 originalContainer.begin(),
480 originalContainer.end(),
481 std::back_inserter(castedContainer),
482 [](const ObjectType& item) { return item.template cast<OtherScalar>(); });
483
484 return castedContainer;
485 }
