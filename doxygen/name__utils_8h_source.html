27template <typename T>
28std::string makeUniqueName(const std::string& requestedName, const T& existingNames) {
29 if (existingNames.find(requestedName) == existingNames.end()) {
30 return requestedName;
31 }
32
33 std::string candidate = requestedName + '_';
34 const std::size_t baseLength = candidate.size();
35 for (std::size_t suffix = 1;; ++suffix) {
36 candidate.resize(baseLength);
37 candidate += std::to_string(suffix);
38 if (existingNames.find(candidate) == existingNames.end()) {
39 return candidate;
40 }
41 }
42}

55template <typename T>
56class ExportedNameRegistry {
57 public:
61 [[nodiscard]] std::string makeExportedName(const std::string& requestedName) const {
62 std::string exportedName = makeUniqueName(requestedName, usedNames_);
63 if (exportedName != requestedName) {
64 MT_LOGW(
65 "Exported name '{}' is already in use; exporting as '{}'.", requestedName, exportedName);
66 }
67 return exportedName;
68 }
69
73 T& insert(const std::string& exportedName, const std::string& sourceName, T entry) {
74 const auto [it, inserted] = defaultExportedNameBySourceName_.emplace(sourceName, exportedName);
75 if (!inserted && exportedName == sourceName) {
76 it->second = exportedName;
77 }
78 usedNames_.insert(exportedName);
79 return entries_.insert_or_assign(exportedName, std::move(entry)).first->second;
80 }
81
85 void reserve(const std::string& exportedName) {
86 usedNames_.insert(exportedName);
87 }
88
92 [[nodiscard]] T* resolve(const std::string& exportedName, const std::string& sourceName) {
93 if (!exportedName.empty()) {
94 T* entry = find(exportedName);
95 MT_THROW_IF(
96 entry == nullptr, "Character '{}' has not been added to the builder", exportedName);
97 return entry;
98 }
99 const auto it = defaultExportedNameBySourceName_.find(sourceName);
100 return it == defaultExportedNameBySourceName_.end() ? nullptr : find(it->second);
101 }
102
104 [[nodiscard]] T* find(const std::string& exportedName) {
105 const auto it = entries_.find(exportedName);
106 return it == entries_.end() ? nullptr : &it->second;
107 }
108
110 [[nodiscard]] std::size_t size() const {
111 return entries_.size();
112 }
113
114 private:
115 std::unordered_map<std::string, T> entries_;
116 std::unordered_map<std::string, std::string> defaultExportedNameBySourceName_;
117 // Every exported name: the names of the entries and the reserved names.
118 std::unordered_set<std::string> usedNames_;
119};
