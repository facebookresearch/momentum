59fx::gltf::Animation& addMomentumAnimationToModel(
60 fx::gltf::Document& model,
61 const std::string& nameIn = "default") {
62 auto animationIter = std::find_if(
63 model.animations.begin(), model.animations.end(), [nameIn](const fx::gltf::Animation& anim) {
64 return nameIn == anim.name;
65 });
66 if (animationIter == model.animations.end()) {
67 model.animations.emplace_back();
68 auto& animation = model.animations.at(model.animations.size() - 1);
69 animation.name = nameIn;
70 return animation;
71 }
72 return *animationIter;
73}

76 fx::gltf::Document& model,
77 const uint32_t accessorId,
78 fx::gltf::BufferView::TargetType targetType) {
79 MT_CHECK(accessorId < model.accessors.size());
80 const auto accessor = model.accessors[accessorId];
81 MT_CHECK(accessor.bufferView < model.bufferViews.size());

87// Add a mesh node with only one primitive (i.e. combined mesh)
88// Use a flag to decide if we should save the vertex color. If a mesh only has default color, we may
89// not want to save it.
90size_t addMeshToModel(fx::gltf::Document& model, const Mesh& mesh, const bool addColor = false) {
91 if (mesh.vertices.empty()) {
92 return std::numeric_limits<size_t>::max();
93 }
94
95 model.meshes.emplace_back();
96 auto& m = model.meshes.back();
97 m.primitives.emplace_back();
98 auto& prim = m.primitives.back();
99
100 // Add faces to mesh
101 if ((!mesh.faces.empty()) && (mesh.faces[0].size() > 0)) {
102 prim.mode = fx::gltf::Primitive::Mode::Triangles;
103 const std::span<const int32_t> faces(&mesh.faces[0][0], mesh.faces.size() * 3);
104 prim.indices = createAccessorBuffer<const int32_t>(model, faces);
105 setBufferView(model, prim.indices, fx::gltf::BufferView::TargetType::ElementArrayBuffer);
106 } else {
107 prim.mode = fx::gltf::Primitive::Mode::Points;
108 }
109
110 // Add vertices to mesh
111 std::vector<Vector3f> verts = mesh.vertices;
112 fromMomentumVec3f(verts);
113 MT_CHECK((!verts.empty()) && (verts[0].size() > 0));
114 prim.attributes["POSITION"] = createAccessorBuffer<const Vector3f>(model, verts, true);
115 setBufferView(model, prim.attributes["POSITION"], fx::gltf::BufferView::TargetType::ArrayBuffer);
116
117 // need min/max for position accessor
118 Box3f bbox;
119 for (const auto& v : verts) {
120 bbox.extend(v);
121 }
122 for (size_t d = 0; d < 3; d++) {
123 model.accessors.at(prim.attributes["POSITION"]).min.emplace_back(bbox.min()[d]);
124 model.accessors.at(prim.attributes["POSITION"]).max.emplace_back(bbox.max()[d]);
125 }
126
127 // add normal attribute
128 if (mesh.normals.size() == mesh.vertices.size()) {
129 prim.attributes["NORMAL"] = createAccessorBuffer<const Vector3f>(model, mesh.normals, true);
130 setBufferView(model, prim.attributes["NORMAL"], fx::gltf::BufferView::TargetType::ArrayBuffer);
131 }
132
133 // add texcoord attribute
134 if (!mesh.texcoords.empty()) {
135 // NOTE: GLTF doesn't support storing multiple texture coordinates per vertex, so instead of
136 // using the standard GLTF texture arrays we store the data in a custom location
137 auto& def = addMomentumExtension(prim.extensionsAndExtras);
138 def["texcoords"] = createAccessorBuffer<const Vector2f>(model, mesh.texcoords, true);
139 setBufferView(model, def["texcoords"], fx::gltf::BufferView::TargetType::ArrayBuffer);
140
141 MT_THROW_IF(mesh.texcoord_faces.empty(), "Mesh has texcoords but no texcoord_faces");
142 const std::span<const int32_t> texFaces(
143 &mesh.texcoord_faces[0][0], mesh.texcoord_faces.size() * 3);
144 def["texfaces"] = createAccessorBuffer<const int32_t>(model, texFaces);
145 setBufferView(model, def["texfaces"], fx::gltf::BufferView::TargetType::ArrayBuffer);
146 }
147
148 if (addColor && !mesh.colors.empty() && mesh.colors.size() != mesh.vertices.size()) {
149 MT_LOGW(
150 "Skipping vertex colors because color count ({}) does not match vertex count ({})",
151 mesh.colors.size(),
152 mesh.vertices.size());
153 }
154 if (mesh.colors.size() == mesh.vertices.size() && addColor) {
155 prim.attributes["COLOR_0"] =
156 createAccessorBuffer<const Vector3b>(model, mesh.colors, true, true);
157 setBufferView(model, prim.attributes["COLOR_0"], fx::gltf::BufferView::TargetType::ArrayBuffer);
158 }
159
160 return model.meshes.size() - 1;
161}

163void addMorphTargetsToModel(
164 fx::gltf::Document& model,
165 const BlendShape& shapes,
166 size_t numShapes,
167 size_t meshIndex) {
168 if (model.meshes.empty() || meshIndex == kInvalidIndex) {
169 MT_LOGW("No mesh is saved in the file when adding blendshapes");
170 return;
171 }
172
173 // assume there is only one primitive on the character mesh.
174 auto& prim = model.meshes.at(model.nodes.at(meshIndex).mesh).primitives.at(0);
175 size_t numShapesAdded = prim.targets.size();
176 if (numShapesAdded >= numShapes) {
177 // enough shapes already exist; nothing to be done.
178 return;
179 }
180 auto& def = addMomentumExtension(prim.extensionsAndExtras);
181 def["shapeNames"] = shapes.getShapeNames();
182
183 const MatrixXf& shapeVecs = shapes.getShapeVectors();
184 for (size_t iShape = numShapesAdded; iShape < numShapes; ++iShape) {
185 // make a copy so we can scale it
186 VectorXf deltas = shapeVecs.col(iShape);
187 fromMomentumVec3f(deltas);
188 prim.targets.emplace_back();
189 auto& attr = prim.targets.back();
190
191 std::span<const Eigen::Vector3f> span(
192 reinterpret_cast<Eigen::Vector3f*>(deltas.data()), static_cast<size_t>(deltas.size() / 3));
193 attr["POSITION"] = createAccessorBuffer<const Vector3f>(model, span, true);
194 setBufferView(model, attr["POSITION"], fx::gltf::BufferView::TargetType::ArrayBuffer);
195
196 // set bbox
197 Box3f bbox;
198 for (const Vector3f& v : span) {
199 bbox.extend(v);
200 }
201 for (size_t d = 0; d < 3; ++d) {
202 model.accessors.at(attr["POSITION"]).min.emplace_back(bbox.min()[d]);
203 model.accessors.at(attr["POSITION"]).max.emplace_back(bbox.max()[d]);
204 }
205 }

208void addMorphWeightsToModel(
209 fx::gltf::Document& model,
210 const Character& character,
211 const float fps,
212 const MatrixXf& modelParams,
213 const size_t meshIndex,
214 const std::string& motionName = "default") {
215 const size_t numFrames = modelParams.cols();
216 if (numFrames == 0) {
217 return;
218 }
219
220 const auto& pt = character.parameterTransform;
221 const size_t numWeights = pt.numBlendShapeParameters();
222 if (numWeights == 0) {
223 return;
224 }
225
226 // validate the mesh node - bounds checking
227 MT_THROW_IF(
228 meshIndex >= model.nodes.size() || model.nodes[meshIndex].mesh < 0 ||
229 static_cast<size_t>(model.nodes[meshIndex].mesh) >= model.meshes.size(),
230 "Invalid mesh index for blendshape animation (meshIndex: {}, nodes size: {})",
231 meshIndex,
232 model.nodes.size());
233
234 // validate the mesh node - original logic
235 if (model.meshes[model.nodes[meshIndex].mesh].primitives.size() != 1 ||
236 model.meshes[model.nodes[meshIndex].mesh].primitives.at(0).targets.size() != numWeights) {
237 MT_LOGW("No valid mesh to add blendshape animation for");
238 return;
239 }
240
241 // extract weight values
242 std::vector<float> timestamps(numFrames, 0);
243 bool useChannel = false;
244 // Important to make sure row vs col is what createSampler expects
245 MatrixXf weights(numWeights, numFrames);
246
247 for (size_t iFrame = 0; iFrame < numFrames; ++iFrame) {
248 const VectorXf& pose = modelParams.col(iFrame);
249 const VectorXf w = extractBlendWeights(pt, ModelParameters(pose)).v;
250 if (w.norm() > 1e-5f) {
251 useChannel = true;
252 }
253 weights.col(iFrame) = w;
254 timestamps[iFrame] = static_cast<float>(iFrame) / fps;
255 }
256 if (!useChannel) {
257 return;
258 }
259
260 // store weight values in gltf document
261 auto& animation = addMomentumAnimationToModel(model, motionName);
262 const auto timestampIdx = createAccessorBuffer<const float>(model, timestamps);
263 if (!timestamps.empty() && !model.accessors.empty() && timestampIdx < model.accessors.size()) {
264 model.accessors[timestampIdx].min.emplace_back(timestamps.front());
265 model.accessors[timestampIdx].max.emplace_back(timestamps.back());
266 }
267
268 animation.channels.emplace_back();
269 auto& channel = animation.channels.back();
270 channel.sampler = createSampler<const float>(
271 model,
272 animation,
273 std::span<const float>(weights.data(), numWeights * numFrames),
274 timestampIdx);
275 channel.target.node = static_cast<int32_t>(meshIndex);
276 channel.target.path = "weights";
277}

309void addActorAnimationToModel(
310 fx::gltf::Document& model,
311 const float fps,
312 std::span<const std::vector<momentum::Marker>> markerSequence,
313 const GltfBuilder::MarkerMesh markerMesh,
314 const std::string& animName) {
315 if (markerSequence.empty()) {
316 return;
317 }
318
319 // create arrays for data
320 const float kLowConfidence = 1e-3;
321 std::map<std::string, size_t> markerNameToIndex;
322 std::vector<std::string> markerNames;
323 std::vector<std::vector<float>> timestamps;
324 std::vector<std::vector<Vector3f>> markerPositions;
325 std::vector<std::vector<Vector3f>> markerConf;
326 for (size_t i = 0; i < markerSequence.size(); i++) {
327 const float timestamp = static_cast<float>(i) / fps;
328 for (const auto& marker : markerSequence[i]) {
329 if (marker.occluded || marker.confidence < kLowConfidence) {
330 continue;
331 }
332
333 // create new array if marker is unknown
334 if (markerNameToIndex.count(marker.name) == 0) {
335 const auto index = timestamps.size();
336 timestamps.emplace_back();
337 markerPositions.emplace_back();
338 markerConf.emplace_back();
339 markerNameToIndex[marker.name] = index;
340 markerNames.emplace_back(marker.name);
341 }
342
343 // get index
344 const auto& index = markerNameToIndex.at(marker.name);
345 MT_THROW_IF(
346 index >= timestamps.size() || index >= markerPositions.size() ||
347 index >= markerConf.size(),
348 "Marker index {} exceeds container size",
349 index);
350 timestamps[index].push_back(timestamp);
351 markerPositions[index].push_back(fromMomentumVec3f(marker.pos.cast<float>()));
352 markerConf[index].push_back(Vector3f::Ones() * marker.confidence);
353 }
354 }
355
356 // add data to gltf document
357 auto& scene = getDefaultScene(model);
358 auto& animation = addMomentumAnimationToModel(model, animName);
359
360 MT_CHECK(
361 (markerNames.size() == timestamps.size()) && (markerNames.size() == markerPositions.size()) &&
362 (markerNames.size() == markerNameToIndex.size()) &&
363 (markerNames.size() == markerConf.size()));
364
365 for (size_t j = 0; j < markerConf.size(); j++) {
366 // create marker
367 const auto nodeIndex = static_cast<int32_t>(model.nodes.size());
368 model.nodes.emplace_back();
369 auto& node = model.nodes.back();
370 scene.nodes.push_back(nodeIndex);
371 node.name = markerNames[j];
372 node.translation = {0.0f, 0.0f, 0.0f};
373 node.scale = {1.0f, 1.0f, 1.0f};
374 node.extensionsAndExtras["extensions"]["FB_momentum"]["type"] = "marker";
375 // for convenience, save confidence as a flat float array so we don't need to parse animation
376 // channels
377 {
378 std::vector<float> confValues;
379 confValues.reserve(markerConf[j].size());
380 for (const auto& v : markerConf[j]) {
381 confValues.push_back(v[0]);
382 }
383 node.extensionsAndExtras["extensions"]["FB_momentum"]["confidence"] = confValues;
384 }
385 switch (markerMesh) {
386 case GltfBuilder::MarkerMesh::UnitCube: {
387 auto newMeshIdx = addMeshToModel(model, kUnitCubeRed, true);
388 if (newMeshIdx < model.meshes.size()) {
389 node.mesh = static_cast<int32_t>(newMeshIdx);
390 }
391 break;
392 }
393 case GltfBuilder::MarkerMesh::None:
394 break;
395 }
396
397 // create animation channel
398 const auto timestampIdx = createAccessorBuffer<const float>(model, timestamps[j]);
399 if (!timestamps[j].empty() && !model.accessors.empty() &&
400 timestampIdx < model.accessors.size()) {
401 model.accessors[timestampIdx].min.emplace_back(timestamps[j].front());
402 model.accessors[timestampIdx].max.emplace_back(timestamps[j].back());
403 }
404
405 // translation
406 animation.channels.emplace_back();
407 auto& tchannel = animation.channels.back();
408 tchannel.sampler =
409 // NOLINTNEXTLINE(facebook-hte-ParameterUncheckedArrayBounds)
410 createSampler<const Vector3f>(model, animation, markerPositions[j], timestampIdx);
411 tchannel.target.node = nodeIndex;
412 tchannel.target.path = "translation";
413
414 // scale -- for visualization purpose only
415 animation.channels.emplace_back();
416 auto& schannel = animation.channels.back();
417 schannel.sampler = createSampler<const Vector3f>(model, animation, markerConf[j], timestampIdx);
418 schannel.target.node = nodeIndex;
419 schannel.target.path = "scale";
420 }
421}

423size_t addMeshToModel(
424 fx::gltf::Document& model,
425 const Character& character,
426 const std::vector<size_t>& jointToNodeMap,
427 const size_t rootNodeIdx) {
428 // write mesh node/skinning data
429 if (character.mesh && character.skinWeights) {
430 // create new mesh node
431 const auto nodeIdx = model.nodes.size();
432 model.nodes.emplace_back();
433 auto& node = model.nodes.back();
434 MT_THROW_IF(
435 rootNodeIdx >= model.nodes.size(),
436 "Root node index {} exceeds model.nodes size {}",
437 rootNodeIdx,
438 model.nodes.size());
439 model.nodes[rootNodeIdx].children.push_back(static_cast<int32_t>(nodeIdx));
440
441 // create model mesh
442 const auto& mesh = *character.mesh;
443 auto newMeshIdx = addMeshToModel(model, mesh, !mesh.colors.empty());
444 if (newMeshIdx >= model.meshes.size()) {
445 // Mesh not valid, skip
446 return kInvalidIndex;
447 }
448 node.mesh = static_cast<int32_t>(newMeshIdx);
449 auto& m = model.meshes[node.mesh];
450 MT_CHECK(m.primitives.size() == 1);
451 auto& prim = m.primitives.back();
452
453 const auto& skin = *character.skinWeights;
454 model.skins.emplace_back();
455 auto& sk = model.skins.back();
456
457 // add skin node
458 for (const auto& ji : jointToNodeMap) {
459 sk.joints.push_back(static_cast<uint32_t>(ji));
460 }
461 sk.skeleton = sk.joints[0];
462
463 node.name = "mesh";
464 node.skin = static_cast<int32_t>(model.skins.size() - 1);
465
466 // add inverse bind matrix
467 std::vector<Matrix4f> ibm;
468 for (const auto& mat : character.inverseBindPose) {
469 auto data = mat;
470 data.translation() = fromMomentumVec3f(data.translation());
471 ibm.emplace_back(data.matrix());
472 }
473 sk.inverseBindMatrices = createAccessorBuffer<const Matrix4f>(model, ibm);
474
475 // add skinning data
476 int weightSets = 1;
477 if (skin.weight.block(0, 4, skin.weight.rows(), 4).maxCoeff() > 0.001f) {
478 // only export two sets of weights if more than one set is being used
479 // only one vertex on the second set would have to have more than 0.1% influence to qualify
480 weightSets = 2;
481 }
482 for (size_t b = 0; b < weightSets; b++) {
483 std::vector<Vector4s> indices(skin.index.rows());
484 std::vector<Vector4f> weights(skin.index.rows());
485 for (size_t i = 0; i < indices.size(); i++) {
486 indices[i] = Vector4s(
487 skin.index(i, b * 4 + 0),
488 skin.index(i, b * 4 + 1),
489 skin.index(i, b * 4 + 2),
490 skin.index(i, b * 4 + 3));
491 // NOLINTNEXTLINE(facebook-hte-LocalUncheckedArrayBounds)
492 weights[i] = Vector4f(
493 skin.weight(i, b * 4 + 0),
494 skin.weight(i, b * 4 + 1),
495 skin.weight(i, b * 4 + 2),
496 skin.weight(i, b * 4 + 3));
497 }
498
499 const auto kIndicesAttribute = std::string("JOINTS_") + std::to_string(b);
500 prim.attributes[kIndicesAttribute] = createAccessorBuffer<const Vector4s>(model, indices);
501 setBufferView(
502 model, prim.attributes[kIndicesAttribute], fx::gltf::BufferView::TargetType::ArrayBuffer);
503
504 const auto kWeightsAttribute = std::string("WEIGHTS_") + std::to_string(b);
505 prim.attributes[kWeightsAttribute] = createAccessorBuffer<const Vector4f>(model, weights);
506 setBufferView(
507 model, prim.attributes[kWeightsAttribute], fx::gltf::BufferView::TargetType::ArrayBuffer);
508 }
509 return nodeIdx;
510 }
511 return kInvalidIndex;
512}

514void addSkeletonStatesToModel(
515 fx::gltf::Document& model,
516 const Character& character,
517 const float fps,
518 std::span<const SkeletonState> skeletonStates,
519 std::span<const size_t> jointToNodeMap,
520 const std::string& motionName = "default") {
521 const auto numFrames = skeletonStates.size();
522 if (numFrames == 0) {
523 return;
524 }
525
526 // check for valid character and stuff before adding things
527 MT_THROW_IF(
528 character.skeleton.joints.empty() || character.parameterTransform.numJointParameters() == 0,
529 "Character is not valid when trying to add skeleton states to model");
530
531 // store parameterized motion into joints for actual animation
532 const auto numJoints = character.skeleton.joints.size();
533 std::vector<float> timestamps(numFrames);
534 std::vector<Vector3<bool>> useChannel(numJoints, Vector3<bool>::Constant(false));
535 std::vector<std::vector<Vector3f>> translation(numJoints, std::vector<Vector3f>(numFrames));
536 std::vector<std::vector<Vector4f>> rotation(numJoints, std::vector<Vector4f>(numFrames));
537 std::vector<std::vector<Vector3f>> scale(numJoints, std::vector<Vector3f>(numFrames));
538
539 // create motion data using the mapped data
540 bool foundAnyChannel = false;
541 for (Eigen::Index i = 0; i < static_cast<Eigen::Index>(numFrames); i++) {
542 for (size_t j = 0; j < numJoints; j++) {
543 auto localT = skeletonStates[i].jointState[j].localTranslation().cast<float>();
544 if (!(localT - character.skeleton.joints[j].translationOffset).isZero(1e-5f)) {
545 useChannel[j][0] = true;
546 foundAnyChannel = true;
547 }
548 translation[j][i] = fromMomentumVec3f(localT);
549
550 auto localR = skeletonStates[i].jointState[j].localRotation().cast<float>();
551 rotation[j][i] = {localR.x(), localR.y(), localR.z(), localR.w()};
552 if (!(localR.toRotationMatrix() - character.skeleton.joints[j].preRotation.toRotationMatrix())
553 .isZero(1e-5f)) {
554 useChannel[j][1] = true;
555 foundAnyChannel = true;
556 }
557
558 float localScale = skeletonStates[i].jointState[j].localScale();
559 scale[j][i] = Vector3f::Ones() * localScale;
560 if (std::fabs(localScale - 1.0f) > 1e-5f) {
561 useChannel[j][2] = true;
562 foundAnyChannel = true;
563 }
564 }
565 timestamps[i] = static_cast<float>(i) / fps;
566 }
567
568 // fx-gltf has a bug where if you write an animation that has no actual
569 // animated channels, it throws an error "Required field not found : channels".
570 // So we'll just make sure there is at least one channel in this case.
571 if (!foundAnyChannel && !useChannel.empty()) {
572 useChannel[0][0] = true;
573 }
574
575 // add data to gltf document
576 auto& animation = addMomentumAnimationToModel(model, motionName);
577 const auto timestampIdx = createAccessorBuffer<const float>(model, timestamps);
578 if (!timestamps.empty() && !model.accessors.empty() && timestampIdx < model.accessors.size()) {
579 model.accessors[timestampIdx].min.emplace_back(timestamps.front());
580 model.accessors[timestampIdx].max.emplace_back(timestamps.back());
581 }
582
583 // only save the animation channels that actually have any useful data
584 for (size_t j = 0; j < numJoints; j++) {
585 if (useChannel[j][0]) {
586 animation.channels.emplace_back();
587 auto& tchannel = animation.channels.back();
588 tchannel.sampler =
589 createSampler<const Vector3f>(model, animation, translation[j], timestampIdx);
590 tchannel.target.node = static_cast<int32_t>(jointToNodeMap[j]);
591 tchannel.target.path = "translation";
592 }
593
594 if (useChannel[j][1]) {
595 animation.channels.emplace_back();

603 animation.channels.emplace_back();

606 schannel.target.node = static_cast<int32_t>(jointToNodeMap[j]);

612void addMotionToModel(
613 fx::gltf::Document& model,
614 const Character& character,
615 const float fps,
616 const MotionParameters& inputMotion,
617 const IdentityParameters& inputIdentity,
618 const std::vector<size_t>& jointToNodeMap,
619 const size_t meshIndex,
620 const bool addExtension = true,
621 const std::string& motionName = "default") {
622 // disentangle input values for storage
623 const auto& parameterNames = std::get<0>(inputMotion);
624 const auto& motion = std::get<1>(inputMotion);
625 const auto& jointNames = std::get<0>(inputIdentity);
626 const auto& identity = std::get<1>(inputIdentity);
627
628 const auto numFrames = motion.cols();
629 if (numFrames == 0) {
630 return;
631 }
632
633 // **** Save animation to Momentum extension ****
634 // add motion if we have poses
635 bool fullAnimation = false;
636 if (addExtension) {
637 auto& def = addMomentumExtension(model.extensionsAndExtras);
638 // store motion array and offsets as metadata
639 def["motion"]["nframes"] = numFrames;
640 if (!parameterNames.empty() && motion.size() > 0) {
641 def["motion"]["parameterNames"] = parameterNames;
642 def["motion"]["poses"] = createAccessorBuffer<const float>(
643 model, std::span<const float>(motion.data(), motion.size()));
644 fullAnimation = true;
645 }
646
647 if (!jointNames.empty() && identity.v.size() > 0) {
648 def["motion"]["jointNames"] = jointNames;
649 def["motion"]["offsets"] = createAccessorBuffer<const float>(
650 model, std::span<const float>(identity.v.data(), identity.v.size()));
651 fullAnimation = true;
652 }
653 } else {
654 // There may be animation outside of the momentum machinery
655 fullAnimation = true;
656 }
657
658 // **** Save animation to gltf ****
659 // add blenshapes (that are used) if available
660 if (meshIndex != kInvalidIndex && character.blendShape != nullptr) {
661 addMorphTargetsToModel(
662 model,
663 *character.blendShape.get(),
664 character.parameterTransform.numBlendShapeParameters(),
665 meshIndex);
666
667 addMorphWeightsToModel(model, character, fps, motion, meshIndex, motionName);
668 }
669
670 // check for valid character and motion before adding joint animation
671 if (!fullAnimation || character.skeleton.joints.empty()) {
672 return;
673 }
674
675 // map data to character
676 const auto inputPoses = mapMotionToCharacter(inputMotion, character);
677 const auto inputOffset = mapIdentityToCharacter(inputIdentity, character);
678
679 // create skeleton states from model params
680 std::vector<momentum::SkeletonState> skeletonStates(numFrames);
681 CharacterParameters params;
682 params.offsets = inputOffset;
683 for (Eigen::Index i = 0; i < numFrames; i++) {
684 params.pose = inputPoses.col(i);
685 const CharacterState state(params, character, false, false, false);
686 skeletonStates[i] = state.skeletonState;
687 }
688
689 // add skeleton states to model
690 addSkeletonStatesToModel(model, character, fps, skeletonStates, jointToNodeMap, motionName);
691}

695 document.nodes.emplace_back();
696 auto& node = document.nodes.back();
697 node.name = name;
698 return nodeIdx;

701std::vector<size_t> addSkeletonToModel(
702 fx::gltf::Document& model,
703 const Character& character,
704 const bool updateExtension = true,
705 size_t modelRootNodeIndex = kInvalidIndex) {
706 // add all joints to node list
707 std::vector<size_t> jointToNodeMap;
708 std::unordered_map<size_t, const JointPhysicalProperties*> physicalPropertiesByJoint;
709 if (updateExtension) {
710 physicalPropertiesByJoint.reserve(character.physicalProperties.size());
711 for (const JointPhysicalProperties& properties : character.physicalProperties) {
712 const size_t jointIndex = resolvePhysicalPropertiesJointIndex(properties, character.skeleton);
713 MT_THROW_IF(
714 jointIndex == kInvalidIndex,
715 "Physical properties entry references unknown joint '{}'",
716 properties.jointName);
717 const auto emplaceResult = physicalPropertiesByJoint.emplace(jointIndex, &properties);
718 MT_THROW_IF(
719 !emplaceResult.second,
720 "Multiple physical properties entries resolve to joint '{}' (index {})",
721 character.skeleton.joints.at(jointIndex).name,
722 jointIndex);
723 }
724 }
725
726 for (size_t i = 0; i < character.skeleton.joints.size(); i++) {
727 const auto& joint = character.skeleton.joints[i];
728
729 // create new node
730 const auto nodeIndex = appendNode(model, joint.name);
731 auto& node = model.nodes[nodeIndex];
732 if (i == 0) {
733 if (modelRootNodeIndex != kInvalidIndex && modelRootNodeIndex < model.nodes.size()) {
734 model.nodes[modelRootNodeIndex].children.push_back(static_cast<int32_t>(nodeIndex));
735 } else {
736 auto& scene = getDefaultScene(model);
737 scene.nodes.push_back(static_cast<uint32_t>(nodeIndex));
738 }
739 }
740
741 // set joint values in node
742 node.rotation = fromMomentumQuaternionf(joint.preRotation);
743 const auto translation = fromMomentumVec3f(joint.translationOffset);
744 node.translation = {translation[0], translation[1], translation[2]};
745
746 if (updateExtension) {
747 auto& extension = addMomentumExtension(node.extensionsAndExtras);
748 extension["type"] = std::string("skeleton_joint");
749 const auto propertiesIt = physicalPropertiesByJoint.find(i);
750 if (propertiesIt != physicalPropertiesByJoint.end()) {
751 jointPhysicalPropertiesToJson(*propertiesIt->second, extension["physicalProperties"]);
752 }
753 }
754
755 // add node as child to parent joint
756 if (joint.parent != kInvalidIndex) {
757 model.nodes.at(jointToNodeMap.at(joint.parent))
758 .children.push_back(static_cast<int32_t>(nodeIndex));
759 }
760
761 // add node to map
762 jointToNodeMap.push_back(nodeIndex);
763 }
764
765 return jointToNodeMap;
766}

768void addCollisionsToModel(
769 fx::gltf::Document& model,
770 const Character& character,
771 const std::vector<size_t>& jointToNodeMap,
772 bool addExtensions) {
773 // add all collision objects to node list
774 if (character.collision) {
775 for (size_t i = 0; i < character.collision->size(); i++) {
776 const auto& col = (*character.collision)[i];
777
778 MT_CHECK(col.parent != kInvalidIndex);
779
780 // create new node
781 const auto nodeIndex = model.nodes.size();
782 model.nodes.emplace_back();
783 auto& node = model.nodes.back();
784
785 // add node as child to parent joint
786 model.nodes.at(jointToNodeMap.at(col.parent))
787 .children.push_back(static_cast<int32_t>(nodeIndex));
788
789 // add values for the collision primitive
790 node.name = character.skeleton.joints.at(col.parent).name + "_col";
791
792 node.rotation = fromMomentumQuaternionf(col.transformation.rotation);
793 const auto translation = fromMomentumVec3f(col.transformation.translation);
794 node.translation = {translation[0], translation[1], translation[2]};
795
796 // add extra properties
797 if (addExtensions) {
798 auto& extension = addMomentumExtension(node.extensionsAndExtras);
799 if (col.type == CollisionPrimitiveType::TaperedCapsule) {
800 extension["type"] = "collision_capsule";
801 extension["length"] = col.length;
802 extension["radius"] = col.radius;
803 } else if (col.type == CollisionPrimitiveType::Ellipsoid) {
804 extension["type"] = "collision_ellipsoid";
805 extension["radii"] = col.ellipsoidRadii;
806 } else if (col.type == CollisionPrimitiveType::Box) {
807 extension["type"] = "collision_box";
808 extension["halfExtents"] = col.boxHalfExtents;
809 } else {
810 MT_THROW(
811 "Unsupported collision primitive type {} while writing glTF collision geometry",
812 static_cast<int>(col.type));
813 }
814 }
815 }
816 }

869void saveDocument(
870 const filesystem::path& filename,
871 fx::gltf::Document& model,
872 const GltfFileFormat fileFormat) {
873 // save model
874 auto deducedFileFormat = fileFormat;
875 if (fileFormat == GltfFileFormat::Auto && filename.extension() == filesystem::path(".gltf")) {
876 deducedFileFormat = GltfFileFormat::Ascii;
877 }
878
879 try {
880 // Check if the parent directory of the file exists. If the parent directory does not exist
881 // and is not an empty string (which means the file is in the current working directory),
882 // create it. This is necessary because fx::gltf::Save throws an exception if the directory
883 // does not exist.
884 const filesystem::path parentDir = filename.parent_path();
885 if (!parentDir.empty() && !filesystem::exists(parentDir)) {
886 filesystem::create_directories(parentDir);
887 }
888
889 switch (deducedFileFormat) {
890 case GltfFileFormat::Ascii: {
891 // we can't embed binary if we writing to ascii format.
892 // so we need to provide exported with a uri name relative to json root.
893 filesystem::path new_filename = filename;
894 new_filename.replace_extension(filesystem::path("glbin"));
895 if (!model.buffers.empty()) {
896 model.buffers.front().uri = new_filename.filename().string();
897 }
898
899 MT_LOGE(
900 "There are some compatibility issues with .gltf files. Not all extension data might be exported correctly.");
901 fx::gltf::Save(model, filename.string(), false);
902 break;
903 }
904 case GltfFileFormat::Auto:
905 [[fallthrough]];
906 case GltfFileFormat::Binary:
907 fx::gltf::Save(model, filename.string(), true);
908 break;
909 }
910 } catch (const std::exception& e) {
911 MT_THROW("Failed to save file to: {}. Error: {}", filename.string(), e.what());
912 } catch (...) {
913 std::rethrow_exception(std::current_exception());
914 }
915
916 MT_THROW_IF(!filesystem::exists(filename), "Unable to save: {}", filename.string());
917}

953void GltfBuilder::addCharacter(
954 const Character& character,
955 const Vector3f& positionOffset /*= Vector3f::Zero()*/,
956 const Quaternionf& rotationOffset /*= Quaternionf::Identity()*/,
957 const FileSaveOptions& options) {
958 if (impl_->characterData.find(character.name) != impl_->characterData.end()) {
959 // Character already exist. Doesn't allow character with the same name to be saved.
960 // #TODO: proper warning
961 return;
962 }
963 auto& scene = getDefaultScene(impl_->document);
964
965 // Add character root node
966 const auto rootNodeIdx = appendNode(impl_->document, character.name);
967 MT_CHECK(
968 rootNodeIdx >= 0 && rootNodeIdx < impl_->document.nodes.size(),
969 "{}: {}",
970 rootNodeIdx,
971 impl_->document.nodes.size());
972 auto& node = impl_->document.nodes[rootNodeIdx];
973 const auto translation = fromMomentumVec3f(positionOffset);
974 node.translation = {translation[0], translation[1], translation[2]};
975 node.rotation = fromMomentumQuaternionf(rotationOffset);
976 scene.nodes.push_back(static_cast<uint32_t>(rootNodeIdx));
977 auto& characterData = impl_->characterData[character.name];
978 characterData.rootIndex = rootNodeIdx;
979
980 // fill with data
981 characterData.nodeMap =
982 addSkeletonToModel(impl_->document, character, options.extensions, rootNodeIdx);
983 if (options.collisions) {
984 addCollisionsToModel(impl_->document, character, characterData.nodeMap, options.extensions);
985 }
986 if (options.locators) {
987 addLocatorsToModel(impl_->document, character, characterData.nodeMap, options.extensions);
988 }
989 if (options.mesh) {
990 characterData.meshIndex =
991 addMeshToModel(impl_->document, character, characterData.nodeMap, rootNodeIdx);
992
993 // Add blendshapes if both addMesh and addBlendShapes are true
994 if (options.blendShapes && character.blendShape != nullptr) {
995 addMorphTargetsToModel(
996 impl_->document,
997 *character.blendShape,
998 character.blendShape->shapeSize(),
999 characterData.meshIndex);
1000 }
1001 }
1002 if (options.extensions) {
1003 auto& def = addMomentumExtension(impl_->document.extensionsAndExtras);
1004 parameterTransformToJson(character, def["transform"]);
1005 parameterLimitsToJson(character, def["parameterLimits"]);
1006 parameterSetsToJson(character, def["parameterSet"]);
1007 poseConstraintsToJson(character, def["poseConstraints"]);
1008 // add metadata if it exists
1009 if (!character.metadata.empty()) {
1010 def["metadata"] = character.metadata;
1011 }
1012 }
