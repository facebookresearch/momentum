39 [[nodiscard]] bool isApprox(
40 const JointT<T>& joint,
41 const T& rotTol = Eps<T>(1e-4, 1e-10),
42 const T& tranTol = Eigen::NumTraits<Scalar>::dummy_precision()) const {
43 if (name != joint.name) {
44 return false;
45 }
46
47 if (parent != joint.parent) {
48 return false;
49 }
50
51 if (!preRotation.toRotationMatrix().isApprox(joint.preRotation.toRotationMatrix(), rotTol)) {
52 return false;
53 }
54
55 if (!translationOffset.isApprox(joint.translationOffset, tranTol)) {
56 return false;
57 }
58
59 return true;
60 }

63 template <typename U>
64 [[nodiscard]] auto cast() const {
65 if constexpr (std::is_same_v<T, U>) {
66 return *this;
67 } else {
68 JointT<U> newJoint;
69 newJoint.name = name;
70 newJoint.parent = parent;
71 newJoint.preRotation = preRotation.template cast<U>();
72 newJoint.translationOffset = translationOffset.template cast<U>();
73 return newJoint;
74 }
75 }
