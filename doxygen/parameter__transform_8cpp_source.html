18bool PoseConstraint::operator==(const PoseConstraint& poseConstraint) const {
19 // Compare parameterIdValue as sets — the vector ordering is not semantically
20 // meaningful (see header comment on `parameterIdValue`).
21 std::map<size_t, float> paramIdToValue1;
22 std::copy(
23 parameterIdValue.begin(),
24 parameterIdValue.end(),
25 std::inserter(paramIdToValue1, paramIdToValue1.begin()));
26
27 std::map<size_t, float> paramIdToValue2;
28 std::copy(
29 poseConstraint.parameterIdValue.begin(),
30 poseConstraint.parameterIdValue.end(),
31 std::inserter(paramIdToValue2, paramIdToValue2.begin()));
32
33 if (paramIdToValue1.size() != paramIdToValue2.size()) {
34 return false;
35 }
36
37 auto pred = [](const auto& l, const auto& r) {
38 return ((l.first == r.first) && isApprox(l.second, r.second));
39 };
40 return std::equal(paramIdToValue1.begin(), paramIdToValue1.end(), paramIdToValue2.begin(), pred);
41}

76template <typename T>
77ParameterTransformT<T> ParameterTransformT<T>::identity(
78 momentum::span<const std::string> jointNames) {
79 ParameterTransformT<T> result;
80 const size_t nJoints = jointNames.size();
81 const size_t nJointParameters = nJoints * kParametersPerJoint;
82 result.offsets.setZero(nJointParameters);
83 result.activeJointParams.setConstant(nJointParameters, true);
84 result.transform.resize(nJointParameters, nJointParameters);
85 result.transform.setIdentity();
86 result.name.resize(nJointParameters);
87 for (size_t i = 0; i < nJoints; i++) {
88 for (size_t j = 0; j < kParametersPerJoint; ++j) {
89 result.name[i * kParametersPerJoint + j] =
90 fmt::format("{}_{}", jointNames[i], kJointParameterNames[j]);
91 }
92 }
93 return result;
94}

96template <typename T>
97VectorX<bool> ParameterTransformT<T>::computeActiveJointParams(const ParameterSet& ps) const {
98 VectorX<bool> result = VectorX<bool>::Constant(this->transform.rows(), false);
99 for (int i = 0; i < this->transform.outerSize(); ++i) {
100 for (typename SparseRowMatrix<T>::InnerIterator it(this->transform, i); it; ++it) {
101 if (ps.test(it.col())) {
102 result[it.row()] = 1;
103 }
104 }
105 }
106 return result;

109template <typename T>
110JointParametersT<T> ParameterTransformT<T>::apply(const ModelParametersT<T>& parameters) const {
111 MT_PROFILE_FUNCTION();
112 MT_CHECK(
113 parameters.size() == transform.cols(),
114 "Size mismatch: The size of the parameters vector '{}' does not match the number of columns in the transform matrix '{}'.",
115 parameters.size(),
116 transform.cols());
117 MT_CHECK(
118 offsets.size() == transform.rows(),
119 "Size mismatch: The size of the offsets vector '{}' does not match the number of rows in the transform matrix '{}'.",
120 offsets.size(),
121 transform.rows());
122 return transform * parameters.v + offsets;
123}

126JointParametersT<T> ParameterTransformT<T>::apply(const CharacterParametersT<T>& parameters) const {
127 MT_PROFILE_FUNCTION();
128 MT_CHECK(
129 parameters.pose.size() == transform.cols(),
130 "{} is not {}",
131 parameters.pose.size(),
132 transform.cols());
133 JointParametersT<T> result = transform * parameters.pose.v;
134 if (parameters.offsets.size() > 0) {
135 MT_CHECK(
136 parameters.offsets.size() == transform.rows(),
137 "{} is not {}",
138 parameters.offsets.size(),
139 transform.rows());
140 result.v += parameters.offsets.v;
141 }
142 return result;

228template <typename T>
229ParameterTransformT<T> mapParameterTransformJoints(
230 const ParameterTransformT<T>& parameterTransform,
231 size_t numTargetJoints,
232 const std::vector<size_t>& jointMapping) {
233 ParameterTransformT<T> mappedTransform;
234
235 // Model parameter list is preserved verbatim; only the joint dimension is remapped.
236 mappedTransform.name = parameterTransform.name;
237
238 mappedTransform.offsets.setZero(numTargetJoints * kParametersPerJoint);
239 mappedTransform.activeJointParams.setConstant(numTargetJoints * kParametersPerJoint, false);
240
241 // Re-index per-joint offsets from the source skeleton into the target skeleton.
242 for (size_t i = 0; i < jointMapping.size(); i++) {
243 if (jointMapping[i] != kInvalidIndex) {
244 for (size_t d = 0; d < kParametersPerJoint; d++) {
245 mappedTransform.offsets[jointMapping[i] * kParametersPerJoint + d] =
246 parameterTransform.offsets(i * kParametersPerJoint + d);
247 }
248 }
249 }
250
251 // Re-index transform rows (joint parameters) from the source skeleton into the
252 // target skeleton; columns (model parameters) are unchanged.
253 std::vector<Eigen::Triplet<float>> triplets;
254 for (int k = 0; k < parameterTransform.transform.outerSize(); ++k) {
255 for (typename SparseRowMatrix<T>::InnerIterator it(parameterTransform.transform, k); it; ++it) {
256 // row encodes (joint index * kParametersPerJoint + parameter type); col is the model
257 // parameter.
258 const int jIndex = static_cast<int>(it.row()) / kParametersPerJoint;
259 const int jOffset = static_cast<int>(it.row()) % kParametersPerJoint;
260
261 const auto mappedJoint = jointMapping[jIndex];
262 if (mappedJoint == kInvalidIndex) {
263 continue;
264 }
265
266 triplets.push_back(
267 Eigen::Triplet<float>(
268 static_cast<int>(mappedJoint * kParametersPerJoint + jOffset),
269 static_cast<int>(it.col()),
270 it.value()));
271
272 // Mark this remapped joint parameter row as driven by some model parameter.
273 mappedTransform.activeJointParams[mappedJoint * kParametersPerJoint + jOffset] = true;
274 }
275 }
276
277 mappedTransform.transform.resize(
278 static_cast<int>(numTargetJoints) * kParametersPerJoint,
279 static_cast<int>(mappedTransform.name.size()));
280
281 mappedTransform.transform.setFromTriplets(triplets.begin(), triplets.end());
282
283 // Pass-through metadata: none of these depend on the joint remapping.
284 mappedTransform.parameterSets = parameterTransform.parameterSets;
285 mappedTransform.poseConstraints = parameterTransform.poseConstraints;
286 mappedTransform.blendShapeParameters = parameterTransform.blendShapeParameters;
287 mappedTransform.faceExpressionParameters = parameterTransform.faceExpressionParameters;
288 mappedTransform.skinnedLocatorParameters = parameterTransform.skinnedLocatorParameters;
289
290 return mappedTransform;
291}

293template <typename T>
294std::tuple<ParameterTransformT<T>, ParameterLimits> subsetParameterTransform(
295 const ParameterTransformT<T>& paramTransformOld,
296 const ParameterLimits& paramLimitsOld,
297 const ParameterSet& paramSet) {
298 const auto nOldParam = paramTransformOld.numAllModelParameters();
299
300 std::vector<size_t> oldParamToNewParam(nOldParam, kInvalidIndex);
301 std::vector<size_t> newParamToOldParam;
302
303 for (Eigen::Index iOldParam = 0; iOldParam < nOldParam; ++iOldParam) {
304 if (!paramSet.test(iOldParam)) {
305 continue;
306 }
307
308 oldParamToNewParam[iOldParam] = newParamToOldParam.size();
309 newParamToOldParam.push_back(iOldParam);
310 }
311
312 const auto nNewParam = newParamToOldParam.size();
313
314 ParameterTransformT<T> paramTransformNew;
315
316 paramTransformNew.name.reserve(nNewParam);
317 for (const auto iOldParam : newParamToOldParam) {
318 paramTransformNew.name.push_back(paramTransformOld.name[iOldParam]);
319 }
320
321 // Offsets are unchanged because they are the size #joints, not #param.
322 paramTransformNew.offsets = paramTransformOld.offsets;
323
324 paramTransformNew.activeJointParams.setConstant(paramTransformOld.numJointParameters(), false);
325
326 // Remap the transformation.
327 using IndexType = typename SparseRowMatrix<T>::Index;
328 std::vector<Eigen::Triplet<T>> tripletsNew;
329 for (int k = 0; k < paramTransformOld.transform.outerSize(); ++k) {
330 for (typename SparseRowMatrix<T>::InnerIterator it(paramTransformOld.transform, k); it; ++it) {
331 const auto iOldParam = it.col();
332 const auto iNewParam = oldParamToNewParam[iOldParam];
333
334 if (iNewParam == kInvalidIndex) {
335 continue;
336 }
337
338 tripletsNew.emplace_back((int)it.row(), (int)iNewParam, it.value());
339 paramTransformNew.activeJointParams[it.row()] = true;
340 }
341 }
342
343 paramTransformNew.transform.resize(paramTransformOld.transform.rows(), (IndexType)nNewParam);
344 paramTransformNew.transform.setFromTriplets(tripletsNew.begin(), tripletsNew.end());
345
346 for (const auto& [name, constraintsOld] : paramTransformOld.poseConstraints) {
347 PoseConstraint pcNew;
348 for (const auto& [idxOld, weight] : constraintsOld.parameterIdValue) {
349 auto idxNew = oldParamToNewParam[idxOld];
350 if (idxNew != kInvalidIndex) {
351 pcNew.parameterIdValue.emplace_back(idxNew, weight);
352 }
353 }
354
355 paramTransformNew.poseConstraints.insert(std::make_pair(name, pcNew));
356 }
357
358 paramTransformNew.blendShapeParameters =
359 Eigen::VectorXi::Constant(paramTransformOld.blendShapeParameters.size(), -1);
360 for (Eigen::Index iBlendShape = 0; iBlendShape < paramTransformOld.blendShapeParameters.size();
361 ++iBlendShape) {
362 auto paramOld = paramTransformOld.blendShapeParameters(iBlendShape);
363 if (paramOld >= 0) {
364 auto paramNew = oldParamToNewParam[paramOld];
365 if (paramNew != kInvalidIndex) {
366 paramTransformNew.blendShapeParameters(iBlendShape) = static_cast<int>(paramNew);
367 }
368 }
369 }
370
371 paramTransformNew.faceExpressionParameters =
372 Eigen::VectorXi::Constant(paramTransformOld.faceExpressionParameters.size(), -1);
373 for (Eigen::Index iBlendShape = 0;
374 iBlendShape < paramTransformOld.faceExpressionParameters.size();
375 ++iBlendShape) {
376 auto paramOld = paramTransformOld.faceExpressionParameters(iBlendShape);
377 if (paramOld >= 0) {
378 auto paramNew = oldParamToNewParam[paramOld];
379 if (paramNew != kInvalidIndex) {
380 paramTransformNew.faceExpressionParameters(iBlendShape) = static_cast<int>(paramNew);
381 }
382 }
383 }
384
385 paramTransformNew.skinnedLocatorParameters =
386 Eigen::VectorXi::Constant(paramTransformOld.skinnedLocatorParameters.size(), -1);
387 for (Eigen::Index iSkinnedLocator = 0;
388 iSkinnedLocator < paramTransformOld.skinnedLocatorParameters.size();
389 ++iSkinnedLocator) {
390 auto paramOld = paramTransformOld.skinnedLocatorParameters(iSkinnedLocator);
391 if (paramOld < 0) {
392 continue;
393 }
394
395 bool anyMapped = false;
396 for (int k = 0; k < 3; ++k) {
397 anyMapped = anyMapped || (oldParamToNewParam[paramOld + k] != kInvalidIndex);
398 }
399
400 if (anyMapped) {
401 auto paramNew = oldParamToNewParam[paramOld];
402 for (int k = 0; k < 3; ++k) {
403 if (oldParamToNewParam[paramOld + k] != paramNew + k) {
404 MT_THROW(
405 "When mapping skinned locator parameters, must retain all 3 parameters for each locator. Locator {} is missing parameter {} ({}).",
406 iSkinnedLocator,
407 paramOld + k,
408 paramTransformOld.name[paramOld + k]);
409 }
410 }
411
412 paramTransformNew.skinnedLocatorParameters(iSkinnedLocator) = gsl::narrow_cast<int>(paramNew);
413 }
414 }
415
416 for (const auto& paramSetOld : paramTransformOld.parameterSets) {
417 ParameterSet paramSetNew;
418 paramSetNew.reset();
419
420 for (size_t iNewParam = 0; iNewParam < nNewParam; ++iNewParam) {
421 const auto iOldParam = newParamToOldParam[iNewParam];
422 paramSetNew.set(iNewParam, paramSetOld.second.test(iOldParam));
423 }
424
425 paramTransformNew.parameterSets.emplace(paramSetOld.first, paramSetNew);
426 }
427
428 ParameterLimits paramLimitsNew;
429 paramLimitsNew.reserve(paramLimitsOld.size());
430 for (const auto& limitOld : paramLimitsOld) {
431 auto limitNew = limitOld;
432 switch (limitOld.type) {
433 case MinMax: {
434 auto& data = limitNew.data.minMax;
435 data.parameterIndex = oldParamToNewParam[data.parameterIndex];
436 if (data.parameterIndex == kInvalidIndex) {
437 continue;
438 }
439 break;
440 }
441
442 case MinMaxJoint: {
443 break;
444 }
445
446 case Linear: {
447 auto& data = limitNew.data.linear;
448 data.referenceIndex = oldParamToNewParam[data.referenceIndex];
449 data.targetIndex = oldParamToNewParam[data.targetIndex];
450 if (data.referenceIndex == kInvalidIndex || data.targetIndex == kInvalidIndex) {
451 continue;
452 }
453 break;
454 }
455
456 case LinearJoint: {
457 break;
458 }
459
460 case HalfPlane: {
461 auto& data = limitNew.data.halfPlane;
462 data.param1 = oldParamToNewParam[data.param1];
463 data.param2 = oldParamToNewParam[data.param2];
464 if (data.param1 == kInvalidIndex || data.param2 == kInvalidIndex) {
465 continue;
466 }
467 break;
468 }
469
470 case Ellipsoid:
471 case MinMaxJointPassive:
472 case LimitTypeCount: {
473 // nothing to do here.
474 break;
475 }
476 }
477
478 paramLimitsNew.push_back(limitNew);
479 }
480
481 return std::make_tuple(paramTransformNew, paramLimitsNew);
482}

538template <typename T>
539bool ParameterTransformT<T>::isApprox(const ParameterTransformT<T>& parameterTransform) const {
540 bool isTransformEqual = false;
541 if (transform.cols() > 0 && transform.rows() > 0 && parameterTransform.transform.cols() > 0 &&
542 parameterTransform.transform.rows() > 0) {
543 isTransformEqual = transform.isApprox(parameterTransform.transform);
544 } else {
545 isTransformEqual = (transform.cols() == parameterTransform.transform.cols()) &&
546 (transform.rows() == parameterTransform.transform.rows());
547 }
548 if (!isTransformEqual) {
549 return false;
550 }
551
552 return (
553 (name == parameterTransform.name) && offsets.isApprox(parameterTransform.offsets) &&
554 activeJointParams.isApprox(parameterTransform.activeJointParams) &&
555 (parameterSets == parameterTransform.parameterSets) &&
556 (poseConstraints == parameterTransform.poseConstraints) &&
557 (blendShapeParameters == parameterTransform.blendShapeParameters) &&
558 (faceExpressionParameters == parameterTransform.faceExpressionParameters) &&
559 (skinnedLocatorParameters == parameterTransform.skinnedLocatorParameters));
560}

562std::tuple<ParameterTransform, ParameterLimits> addBlendShapeParameters(
563 ParameterTransform paramTransform,
564 ParameterLimits paramLimits,
565 Eigen::Index nBlendShapes) {
566 // First, strip out any existing blend shape parameters, to make sure this
567 // operation is idempotent.
568 std::tie(paramTransform, paramLimits) = subsetParameterTransform(
569 paramTransform, paramLimits, ~paramTransform.getBlendShapeParameters());
570
571 // Now add in the additional parameters:
572 paramTransform.blendShapeParameters.setZero(nBlendShapes);
573 for (Eigen::Index iBlend = 0; iBlend < nBlendShapes; ++iBlend) {
574 std::ostringstream oss;
575 oss << "blend_" << iBlend;
576 paramTransform.blendShapeParameters(iBlend) = (int)paramTransform.name.size();
577 paramTransform.name.push_back(oss.str());
578 }
579
580 paramTransform.transform.conservativeResize(
581 paramTransform.transform.rows(), paramTransform.name.size());
582 paramTransform.transform.makeCompressed();
583
584 return {paramTransform, paramLimits};
585}

612std::tuple<ParameterTransform, ParameterLimits> addSkinnedLocatorParameters(
613 ParameterTransform paramTransform,
614 ParameterLimits paramLimits,
615 const std::vector<bool>& activeLocators,
616 const std::vector<std::string>& locatorNames) {
617 // First, strip out any existing blend shape parameters, to make sure this
618 // operation is idempotent.
619 std::tie(paramTransform, paramLimits) = subsetParameterTransform(
620 paramTransform, paramLimits, ~paramTransform.getSkinnedLocatorParameters());
621
622 const auto nLocators = activeLocators.size();
623 MT_CHECK(locatorNames.empty() || locatorNames.size() == nLocators);
624 paramTransform.skinnedLocatorParameters.resize(nLocators);
625 paramTransform.skinnedLocatorParameters.setConstant(-1);
626
627 ParameterSet paramSet;
628
629 // Now add in the additional parameters:
630 for (size_t iLocator = 0; iLocator < activeLocators.size(); ++iLocator) {
631 if (!activeLocators[iLocator]) {
632 continue;
633 }
634
635 paramTransform.skinnedLocatorParameters(iLocator) = (int)paramTransform.name.size();
636
637 auto locatorName =
638 iLocator < locatorNames.size() ? locatorNames[iLocator] : std::to_string(iLocator);
639 std::array<const char*, 3> dirNames{"x", "y", "z"};
640 for (const auto& d : dirNames) {
641 paramSet.set(paramTransform.name.size());
642 std::ostringstream oss;
643 oss << "locator_" << locatorName << "_" << d;
644 paramTransform.name.push_back(oss.str());
645 }
646 }
647
648 paramTransform.parameterSets["skinnedLocators"] = paramSet;
649 paramTransform.transform.conservativeResize(
650 paramTransform.transform.rows(), paramTransform.name.size());
651 paramTransform.transform.makeCompressed();
652
653 return {paramTransform, paramLimits};
654}
