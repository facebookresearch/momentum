27template <typename T>
28void CollisionGeometryStateT<T>::updatePrimitive(
29 const SkeletonStateT<T>& skeletonState,
30 const CollisionGeometry& collisionGeometry,
31 const size_t primitiveIndex) {
32 const auto& primitive = collisionGeometry[primitiveIndex];
33 const TransformT<T> parentTransform = (primitive.parent == kInvalidIndex)
34 ? TransformT<T>()
35 : skeletonState.jointState[primitive.parent].transform;
36 const TransformT<T> transform = parentTransform * primitive.transformation.cast<T>();
37 type[primitiveIndex] = primitive.type;
38 origin[primitiveIndex] = transform.translation;
39 orientation[primitiveIndex] = transform.rotation;
40
41 if (primitive.type == CollisionPrimitiveType::TaperedCapsule) {
42 direction[primitiveIndex].noalias() = transform.getAxisX() * transform.scale * primitive.length;
43 radius[primitiveIndex].noalias() = primitive.radius.cast<T>() * parentTransform.scale;
44 delta[primitiveIndex] = radius[primitiveIndex][1] - radius[primitiveIndex][0];
45 ellipsoidRadii[primitiveIndex].setZero();
46 boxHalfExtents[primitiveIndex].setZero();
47 return;
48 }
49
50 if (primitive.type == CollisionPrimitiveType::Ellipsoid) {
51 direction[primitiveIndex].setZero();
52 radius[primitiveIndex].setZero();
53 delta[primitiveIndex] = T(0);
54 ellipsoidRadii[primitiveIndex].noalias() =
55 primitive.ellipsoidRadii.cast<T>() * parentTransform.scale;
56 boxHalfExtents[primitiveIndex].setZero();
57 return;
58 }
59
60 if (primitive.type == CollisionPrimitiveType::Box) {
61 direction[primitiveIndex].setZero();
62 radius[primitiveIndex].setZero();
63 delta[primitiveIndex] = T(0);
64 ellipsoidRadii[primitiveIndex].setZero();
65 boxHalfExtents[primitiveIndex].noalias() =
66 primitive.boxHalfExtents.cast<T>() * parentTransform.scale;
67 return;
68 }
69}

71template <typename T>
72void CollisionGeometryStateT<T>::update(
73 const SkeletonStateT<T>& skeletonState,
74 const CollisionGeometry& collisionGeometry) {
75 const size_t numElements = collisionGeometry.size();
76 resize(numElements);
77
78 // TODO: This per-primitive loop is independent across iterations and could be parallelized.
79 for (size_t i = 0; i < numElements; ++i) {
80 updatePrimitive(skeletonState, collisionGeometry, i);
81 }
82}
