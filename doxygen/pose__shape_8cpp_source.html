15std::vector<Vector3f> PoseShape::compute(const SkeletonState& state) const {
16 MT_CHECK(
17 baseShape.size() == shapeVectors.rows(),
18 "{} is not {}",
19 baseShape.size(),
20 shapeVectors.rows());
21
22 VectorXf coefficients(shapeVectors.cols());
23
24 // If baseJoint is out of range, fall back to baseRot directly (no relative rotation).
25 const Quaternionf base = (baseJoint < state.jointState.size())
26 ? baseRot * state.jointState.at(baseJoint).rotation().inverse()
27 : baseRot;
28
29 // Pack each driving joint's rotation (relative to base) as a 4-vector of quaternion
30 // coefficients (x, y, z, w). Out-of-range joints leave their segment uninitialized.
31 // TODO: zero-initialize `coefficients` so out-of-range joint indices don't read garbage.
32 for (size_t i = 0; i < jointMap.size(); i++) {
33 const auto& jid = jointMap[i];
34 if (jid < state.jointState.size()) {
35 coefficients.segment<4>(i * 4) = (base * state.jointState.at(jid).rotation()).coeffs();
36 }
37 }
38
39 std::vector<Vector3f> output(baseShape.size() / 3);
40 if (output.empty()) {
41 return output;
42 }
43 Map<VectorXf> outputVec(&output[0][0], output.size() * 3);
44
45 outputVec = baseShape + shapeVectors * coefficients;
46
47 return output;
48}
