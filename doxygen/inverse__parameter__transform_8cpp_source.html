18template <typename T>
19InverseParameterTransformT<T>::InverseParameterTransformT(
20 const ParameterTransformT<T>& paramTransform)
21 : transform(paramTransform.transform),
22 pseudoInverseMatrix(pseudoInverse(MatrixX<T>(paramTransform.transform))),
23 offsets(paramTransform.offsets) {}
24
25template <typename T>
26CharacterParametersT<T> InverseParameterTransformT<T>::apply(
27 const JointParametersT<T>& jointParameters) const {
28 MT_CHECK(
29 jointParameters.size() == transform.rows(),
30 "{} is not {}",
31 jointParameters.size(),
32 transform.rows());
33 MT_CHECK(offsets.size() == transform.rows(), "{} is not {}", offsets.size(), transform.rows());
34 CharacterParametersT<T> result;
35 result.pose = pseudoInverseMatrix * (jointParameters.v - offsets);
36 result.offsets = jointParameters.v - transform * result.pose.v;
37 return result;
38}

43ModelParameters applyModelParameterScales(
44 const ParameterTransform& parameterTransform,
45 MatrixXf& motion,
46 const JointParameters& jointIdentity) {
47 const auto scalingParams = parameterTransform.getScalingParameters();
48
49 // Check if motion already contains non-zero scale parameters
50 bool hasScaleParametersInMotion = false;
51 for (Eigen::Index iFrame = 0; iFrame < motion.cols() && !hasScaleParametersInMotion; ++iFrame) {
52 for (Eigen::Index iParam = 0; iParam < parameterTransform.numAllModelParameters(); ++iParam) {
53 if (scalingParams.test(iParam) && motion(iParam, iFrame) != 0) {
54 hasScaleParametersInMotion = true;
55 break;
56 }
57 }
58 }
59
60 // Convert joint identity parameters to model identity parameters
61 ModelParameters modelIdentity = ModelParameters::Zero(parameterTransform.numAllModelParameters());
62 if (jointIdentity.size() > 0 && !jointIdentity.v.isZero()) {
63 if (hasScaleParametersInMotion) {
64 MT_LOGW("Motion already contains scale parameters, not adding joint identity parameters.");
65 }
66 const auto scalingTransform = parameterTransform.simplify(scalingParams);
67 const ModelParameters scaleParametersSubset =
68 InverseParameterTransform(scalingTransform).apply(jointIdentity).pose;
69
70 MT_CHECK(
71 static_cast<Eigen::Index>(parameterTransform.name.size()) ==
72 parameterTransform.numAllModelParameters(),
73 "Parameter name count ({}) does not match model parameter count ({})",
74 parameterTransform.name.size(),
75 parameterTransform.numAllModelParameters());
76 for (Eigen::Index iParam = 0; iParam < parameterTransform.numAllModelParameters(); iParam++) {
77 if (scalingParams.test(iParam)) {
78 MT_THROW_IF(
79 static_cast<size_t>(iParam) >= parameterTransform.name.size(),
80 "Parameter index {} exceeds parameter name count {}",
81 iParam,
82 parameterTransform.name.size());
83 auto subsetParamIdx =
84 scalingTransform.getParameterIdByName(parameterTransform.name[iParam]);
85 MT_THROW_IF(
86 subsetParamIdx >= static_cast<size_t>(scaleParametersSubset.size()),
87 "Subset parameter index {} exceeds scaleParametersSubset size {}",
88 subsetParamIdx,
89 scaleParametersSubset.size());
90 modelIdentity[iParam] = scaleParametersSubset(subsetParamIdx);
91
92 // Add the identity parameters into the motion.
93 if (!hasScaleParametersInMotion) {
94 for (Eigen::Index jFrame = 0; jFrame < motion.cols(); ++jFrame) {
95 motion(iParam, jFrame) += scaleParametersSubset(subsetParamIdx);
96 }
97 }
98 }
99 }
100 }
101
102 return modelIdentity;
103}
