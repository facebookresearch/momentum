14template <typename T>
15std::vector<PlaneDataT<T>> createFloorConstraints(
16 const std::string& prefix,
17 const LocatorList& locators,
18 const Vector3<T>& floorNormal,
19 const T& floorOffset,
20 const float weight) {
21 std::vector<PlaneDataT<T>> result;
22 for (const auto& loc : locators) {
23 if (loc.name.find(prefix) == 0) {
24 result.emplace_back(
25 PlaneDataT<T>(
26 loc.offset.cast<T>(),
27 floorNormal,
28 floorOffset,
29 loc.parent,
30 loc.weight * weight,
31 loc.name));
32 }
33 }
34 return result;
35}

50template <typename T>
51void PlaneErrorFunctionT<T>::evalFunction(
52 const size_t constrIndex,
53 const JointStateT<T>& state,
54 Vector<T, 1>& f,
55 std::array<Vector3<T>, 1>& v,
56 std::array<Eigen::Matrix<T, 1, 3>, 1>& dfdv) const {
57 MT_PROFILE_FUNCTION();
58
59 const PlaneDataT<T>& constr = this->constraints_[constrIndex];
60 v[0] = state.transform * constr.offset;
61 T val = v[0].dot(constr.normal) - constr.d;
62 if (halfPlane_ && val > T(0)) {
63 val = T(0);
64 }
65 f[0] = val;
66
67 dfdv[0].setZero();
68 if (!halfPlane_ || val < T(0)) {
69 dfdv[0] = constr.normal.transpose();
70 }
71}
