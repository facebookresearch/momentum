47template <typename T>
48void CollisionErrorFunctionT<T>::updateCollisionPairs(bool filterRestPoseOverlaps) {
49 validPairs_.clear();
50
51 const SkeletonStateT<T> state(
52 this->parameterTransform_.zero().template cast<T>(), this->skeleton_);
53 collisionState_.update(state, collisionGeometry_);
54
55 const auto n = collisionGeometry_.size();
56
57 aabbs_.resize(n);
58 for (size_t i = 0; i < n; ++i) {
59 aabbs_[i].id = gsl::narrow_cast<axel::Index>(i);
60 updateAabb(aabbs_[i], collisionState_, i);
61 }
62
63 if (n == 0) {
64 return;
65 }
66
67 for (size_t i = 0; i < n; ++i) {
68 for (size_t j = i + 1; j < n; ++j) {
69 if (isValidCollisionPair(
70 collisionState_, collisionGeometry_, this->skeleton_, i, j, filterRestPoseOverlaps)) {
71 const size_t lca = this->skeleton_.commonAncestor(
72 collisionGeometry_[i].parent, collisionGeometry_[j].parent);
73 validPairs_.push_back({i, j, lca});
74 }
75 }
76 }
77}

156template <typename T>
157std::vector<Vector2i> CollisionErrorFunctionT<T>::getCollisionPairs() const {
158 std::vector<Vector2i> collidingPairs;
159 for (const auto& pair : validPairs_) {
160 CollisionOverlapResultT<T> result;
161 if (checkCollision(collisionState_, pair.indexA, pair.indexB, result)) {
162 collidingPairs.emplace_back(static_cast<int>(pair.indexA), static_cast<int>(pair.indexB));
163 }
164 }
165 return collidingPairs;

168template <typename T>
169double CollisionErrorFunctionT<T>::getError(
170 const ModelParametersT<T>& /* params */,
171 const SkeletonStateT<T>& state,
172 const MeshStateT<T>& /* meshState */) {
173 if (state.jointState.empty()) {
174 return 0.0;
175 }
176
177 MT_CHECK(
178 state.jointParameters.size() ==
179 gsl::narrow<Eigen::Index>(this->skeleton_.joints.size() * kParametersPerJoint));
180
181 computeBroadPhase(state);
182
183 double error = 0;
184 size_t collidingCount = 0;
185 for (const auto& pair : validPairs_) {
186 CollisionOverlapResultT<T> result;
187 if (!checkCollision(collisionState_, pair.indexA, pair.indexB, result)) {
188 continue;
189 }
190 collidingCount++;
191 error += sqr(result.overlap) * kCollisionWeight * this->weight_;
192 }
193
194 MT_LOGD(
195 "CollisionError: {} colliding / {} valid pairs, error={:.8f}, weight={:.4f}",
196 collidingCount,
197 validPairs_.size(),
198 error,
199 this->weight_);
200
201 return error;
202}

204template <typename T>
205double CollisionErrorFunctionT<T>::getGradient(
206 const ModelParametersT<T>& /* params */,
207 const SkeletonStateT<T>& state,
208 const MeshStateT<T>& /* meshState */,
209 Ref<VectorX<T>> gradient) {
210 if (state.jointState.empty()) {
211 return 0.0;
212 }
213
214 MT_CHECK(
215 state.jointParameters.size() ==
216 gsl::narrow<Eigen::Index>(this->skeleton_.joints.size() * kParametersPerJoint));
217
218 computeBroadPhase(state);
219
220 double error = 0;
221 for (const auto& pair : validPairs_) {
222 CollisionOverlapResultT<T> result;
223 if (!checkCollision(collisionState_, pair.indexA, pair.indexB, result)) {
224 continue;
225 }
226
227 error += sqr(result.overlap) * kCollisionWeight * this->weight_;
228
229 const Vector3<T> direction = result.positionA - result.positionB;
230 const T wgt = -T(2) * kCollisionWeight * this->weight_ * (result.overlap / result.distance);
231
232 accumulateGradientAlongChain(
233 state,
234 result.positionA,
235 direction,
236 wgt,
237 collisionGeometry_[pair.indexA].parent,
238 pair.commonAncestor,
239 gradient,
240 -result.distance * result.radiusA * ln2<T>());
241 accumulateGradientAlongChain(
242 state,
243 result.positionB,
244 direction,
245 -wgt,
246 collisionGeometry_[pair.indexB].parent,
247 pair.commonAncestor,
248 gradient,
249 result.distance * result.radiusB * ln2<T>());
250
251 // Scale derivatives above the common ancestor: translation and rotation derivatives cancel
252 // above the LCA (both capsule endpoints move identically), but scale derivatives do not
253 // because the two endpoints are at different distances from the scaled joint, and scaling
254 // also changes capsule radii.
255 {
256 const T netScaleVal =
257 (direction.squaredNorm() - result.distance * (result.radiusA + result.radiusB)) *
258 ln2<T>() * wgt;
259 size_t ancestorIndex = pair.commonAncestor;
260 while (ancestorIndex != kInvalidIndex) {
261 const size_t paramIndex = ancestorIndex * kParametersPerJoint;
262 if (this->activeJointParams_[paramIndex + 6]) {
263 gradient_jointParams_to_modelParams(
264 netScaleVal, paramIndex + 6, this->parameterTransform_, gradient);
265 }
266 ancestorIndex = this->skeleton_.joints[ancestorIndex].parent;
267 }
268 }
269 }
270
271 return error;
272}

274template <typename T>
275double CollisionErrorFunctionT<T>::getJacobian(
276 const ModelParametersT<T>& /* params */,
277 const SkeletonStateT<T>& state,
278 const MeshStateT<T>& /* meshState */,
279 Ref<MatrixX<T>> jacobian,
280 Ref<VectorX<T>> residual,
281 int& usedRows) {
282 MT_PROFILE_FUNCTION();
283 MT_CHECK(
284 jacobian.rows() >= gsl::narrow<Eigen::Index>(getJacobianSize()),
285 "Jacobian rows mismatch: Actual {}, Expected {}",
286 jacobian.rows(),
287 gsl::narrow<Eigen::Index>(getJacobianSize()));
288 MT_CHECK(
289 residual.rows() >= gsl::narrow<Eigen::Index>(getJacobianSize()),
290 "Residual rows mismatch: Actual {}, Expected {}",
291 residual.rows(),
292 gsl::narrow<Eigen::Index>(getJacobianSize()));
293
294 if (state.jointState.empty()) {
295 return 0.0;
296 }
297
298 MT_CHECK(
299 state.jointParameters.size() ==
300 gsl::narrow<Eigen::Index>(this->skeleton_.joints.size() * kParametersPerJoint));
301
302 computeBroadPhase(state);
303
304 const T wgt = std::sqrt(kCollisionWeight * this->weight_);
305 int pos = 0;
306 double error = 0;
307
308 for (const auto& pair : validPairs_) {
309 CollisionOverlapResultT<T> result;
310 if (!checkCollision(collisionState_, pair.indexA, pair.indexB, result)) {
311 continue;
312 }
313
314 error += sqr(result.overlap) * kCollisionWeight * this->weight_;
315
316 const Vector3<T> direction = result.positionA - result.positionB;
317 const T fac = wgt / result.distance;
318 const int row = pos++;
319
320 accumulateJacobianAlongChain(
321 state,
322 result.positionA,
323 direction,
324 -fac,
325 collisionGeometry_[pair.indexA].parent,
326 pair.commonAncestor,
327 jacobian,
328 row,
329 -result.distance * result.radiusA * ln2<T>());
330 accumulateJacobianAlongChain(
331 state,
332 result.positionB,
333 direction,
334 fac,
335 collisionGeometry_[pair.indexB].parent,
336 pair.commonAncestor,
337 jacobian,
338 row,
339 result.distance * result.radiusB * ln2<T>());
340
341 // Scale derivatives above the common ancestor
342 {
343 const T netScaleVal = -fac * ln2<T>() * direction.squaredNorm() +
344 wgt * (result.radiusA + result.radiusB) * ln2<T>();
345 size_t ancestorIndex = pair.commonAncestor;
346 while (ancestorIndex != kInvalidIndex) {
347 const size_t paramIndex = ancestorIndex * kParametersPerJoint;
348 if (this->activeJointParams_[paramIndex + 6]) {
349 jacobian_jointParams_to_modelParams(
350 netScaleVal, paramIndex + 6, row, this->parameterTransform_, jacobian);
351 }
352 ancestorIndex = this->skeleton_.joints[ancestorIndex].parent;
353 }
354 }
355
356 residual(row) = result.overlap * wgt;
357 }
358
359 usedRows = pos;
360 return error;
361}
