14template <typename T>
15void AimDistErrorFunctionT<T>::evalFunction(
16 const size_t constrIndex,
17 const JointStateT<T>& state,
18 Vector3<T>& f,
19 std::array<Vector3<T>, 2>& v,
20 std::array<Eigen::Matrix3<T>, 2>& dfdv) const {
21 MT_PROFILE_FUNCTION();
22
23 const AimDataT<T>& constr = this->constraints_[constrIndex];
24 v[0] = state.transform * constr.localPoint;
25 v[1] = state.rotation() * constr.localDir;
26 const Vector3<T> tgtVec = constr.globalTarget - v[0];
27 const T projLength = v[1].dot(tgtVec);
28
29 // f = (globalTarget - point).dot(srcDir) * srcDir - (globalTarget - point)
30 f = projLength * v[1] - tgtVec;
31 // df/d(point) = I - outterProd(srcDir, srcDir)
32 dfdv[0].setIdentity();
33 dfdv[0].noalias() -= v[1] * v[1].transpose();
34 // df/d(dir) = projLength * I + outterProd(srcDir, tgtVec)
35 dfdv[1].noalias() = v[1] * tgtVec.transpose();
36 dfdv[1].diagonal().array() += projLength;
37}

39template <typename T>
40void AimDirErrorFunctionT<T>::evalFunction(
41 const size_t constrIndex,
42 const JointStateT<T>& state,
43 Vector3<T>& f,
44 std::array<Vector3<T>, 2>& v,
45 std::array<Eigen::Matrix3<T>, 2>& dfdv) const {
46 MT_PROFILE_FUNCTION();
47
48 const AimDataT<T>& constr = this->constraints_[constrIndex];
49 v[0] = state.transform * constr.localPoint;
50 v[1] = state.rotation() * constr.localDir;
51 const Vector3<T> tgtVec = constr.globalTarget - v[0];
52 const T tgtNorm = tgtVec.norm();
53 Vector3<T> tgtDir = Vector3<T>::Zero();
54 if (tgtNorm > 1e-16) {
55 tgtDir = tgtVec / tgtNorm;
56 }
57
58 // f = srcDir - (globalTarget - point).normalize()
59 f = v[1] - tgtDir;
60 // df/d(point) = (I - outterProd(tgtDir, tgtDir)) / tgtNorm
61 dfdv[0].setZero();
62 if (tgtNorm > 1e-16) {
63 dfdv[0].noalias() -= (tgtDir * tgtDir.transpose()) / tgtNorm;
64 dfdv[0].diagonal().array() += T(1) / tgtNorm;
65 }
66 dfdv[1].setIdentity();
67}
