59template <typename T>
60TransformT<T> TransformT<T>::fromMatrix(const Matrix4<T>& other) {
61 TransformT<T> result;
62 result.translation = other.template topRightCorner<3, 1>();
63 // Recover scale from the norm of column 0. This relies on the upper-left 3x3 being
64 // s*R where R is orthonormal (uniform scale, no shear, no reflection). For non-uniform
65 // scaling, shear, or negative determinant, the resulting rotation will be incorrect.
66 const auto& scaledR = other.template topLeftCorner<3, 3>();
67 result.scale = scaledR.col(0).norm();
68 // Guard against near-zero or near-infinite scale before dividing.
69 MT_CHECK(result.scale >= Eps<T>(), "Scale is too small: {}", result.scale);
70 MT_CHECK(result.scale < T(1) / Eps<T>(), "Inverse scale is too small: {}", result.scale);
71 // TODO: Detect non-uniform scale / shear / reflection (e.g. via SVD) and either error
72 // or project to the closest rigid+uniform-scale transform; current quaternion ctor
73 // silently ignores non-orthonormal input.
74 result.rotation = scaledR / result.scale;
75 return result;
76}

92template <typename T>
93TransformT<T> TransformT<T>::inverse() const {
94 // (translate(t) * rotation(R) * scale(s)).inv() =
95 // scale(s).inv() * rotation(R).inv() * translate(t).inv()
96 // = scale(1/s) * rotation(invR) * translate(-t)
97 // = translate(-invR*s*t) * rotation(invR) * scale(1/s)
98 const Quaternion<T> invRot = rotation.inverse();
99 const T invScale = T(1) / scale;
100 return TransformT<T>(-invScale * (invRot * translation), invRot, invScale);
101}

103template <typename T>
104TransformT<T> blendTransforms(
105 momentum::span<const TransformT<T>> transforms,
106 momentum::span<const T> weights) {
107 assert(transforms.size() == weights.size());
108 if (transforms.empty()) {
109 return TransformT<T>();
110 }
111
112 // Find average rotation by means described in
113 // https://stackoverflow.com/questions/12374087/average-of-multiple-quaternions
114 // http://www.acsu.buffalo.edu/~johnc/ave_quat07.pdf
115 //
116 // i.e. Stack quaternion coeffs in Q, compute M = Q^T x Q, and yield the eigenvector corresponding
117 // to the largest eigenvalue as the average rotation. This handles the q vs -q double-cover
118 // ambiguity correctly (since q*q^T == (-q)*(-q)^T) without needing manual sign alignment.
119 Matrix4<T> QtQ = Matrix4<T>::Zero();
120
121 const auto n = transforms.size();
122 for (size_t i = 0; i < n; ++i) {
123 const auto& q = transforms[i].rotation;
124 QtQ += weights[i] * (q.coeffs() * q.coeffs().transpose());
125 }
126
127 const Eigen::SelfAdjointEigenSolver<Matrix4<T>> eigDecomp(QtQ);
128 assert(
129 eigDecomp.eigenvalues()[2] <=
130 eigDecomp.eigenvalues()[3]); // Eigen is supposed to sort smallest to largest.
131 Vector4<T> avgcoeffs = eigDecomp.eigenvectors().col(3);
132 assert(std::abs(avgcoeffs.norm() - 1) < 1e-4); // should be guaranteed by the eigensolver.
133 Quaternion<T> rot(avgcoeffs);
134
135 Vector3<T> pos = Vector3<T>::Zero();
136 T weightSum = 0;
137 T scale = 0;
138 for (size_t i = 0; i < n; ++i) {
139 pos += weights[i] * transforms[i].translation;
140 scale += weights[i] * transforms[i].scale;
141 weightSum += weights[i];
142 }
143
144 // TODO: When weightSum is zero (e.g. all weights zero, or weights cancel), pos and scale
145 // are left at 0 which produces a degenerate transform; consider returning identity or
146 // the first input instead.
147 if (weightSum != 0) {
148 pos /= weightSum;
149 scale /= weightSum;
150 }
151
152 return TransformT<T>{pos, rot, scale};
153}

155template <typename T>
156TransformT<T> slerp(const TransformT<T>& t1, const TransformT<T>& t2, T weight) {
157 return TransformT<T>(
158 t1.translation + weight * (t2.translation - t1.translation),
159 t1.rotation.slerp(weight, t2.rotation),
160 t1.scale + weight * (t2.scale - t1.scale));
161}
