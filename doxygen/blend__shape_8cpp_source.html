24template <typename T>
25void BlendShape::computeShape(
26 const BlendWeightsT<T>& coefficients,
27 std::vector<Eigen::Vector3<T>>& output) const {
28 MT_CHECK(
29 static_cast<Eigen::Index>(baseShape_.size()) * 3 == shapeVectors_.rows(),
30 "{} is not {}",
31 baseShape_.size() * 3,
32 shapeVectors_.rows());
33
34 output.resize(baseShape_.size());
35 Eigen::Map<Eigen::VectorX<T>> outputVec(&output[0][0], output.size() * 3);
36 const Eigen::Map<const Eigen::VectorXf> baseVec(&baseShape_[0][0], baseShape_.size() * 3);
37
38 outputVec = baseVec.template cast<T>() + computeDeltas<T>(coefficients);
39}

41VectorXf BlendShape::estimateCoefficients(
42 momentum::span<const Vector3f> vertices,
43 const float regularization,
44 const VectorXf& weights) const {
45 MT_CHECK(
46 static_cast<Eigen::Index>(baseShape_.size()) * 3 == shapeVectors_.rows(),
47 "{} is not {}",
48 baseShape_.size() * 3,
49 shapeVectors_.rows());
50 MT_CHECK(
51 vertices.size() == baseShape_.size(), "{} is not {}", vertices.size(), baseShape_.size());
52
53 const Map<const VectorXf> baseVec(&baseShape_[0][0], baseShape_.size() * 3);
54 const Map<const VectorXf> vertexVec(&vertices[0][0], vertices.size() * 3);
55
56 // Unweighted path: reuse the cached SVD of shapeVectors_ across calls. Note that
57 // `regularization` is ignored here — the SVD-based pseudoinverse already provides
58 // implicit regularization on near-zero singular values.
59 // TODO: regularization is silently ignored on this branch; either honor it (e.g.
60 // damped least squares) or document it on the header.
61 if (weights.size() != static_cast<Eigen::Index>(vertices.size())) {
62 if (!factorizationValid_) {
63 factorization_ =
64 Eigen::JacobiSVD<MatrixXf>(shapeVectors_, Eigen::ComputeThinU | Eigen::ComputeThinV);
65 factorizationValid_ = true;
66 }
67
68 return factorization_.solve(vertexVec - baseVec);
69 } else {
70 // Weighted path: solve the regularized weighted least-squares problem
71 // min ||W (shapeVectors_ x - (vertexVec - baseVec))||^2 + regularization^2 ||x||^2
72 // by stacking the weighted system on top of an identity block (Tikhonov form)
73 // and solving the resulting normal equations with LDLT. Each per-vertex weight is
74 // replicated 3x to cover x/y/z components. The cached SVD is not used here and is
75 // not invalidated, since shapeVectors_ itself is unchanged.
76 VectorXf wts(weights.size() * 3);
77 Eigen::Map<VectorXf, 0, Eigen::Stride<1, 3>>(&wts(0), weights.size()) = weights;
78 Eigen::Map<VectorXf, 0, Eigen::Stride<1, 3>>(&wts(1), weights.size()) = weights;
79 Eigen::Map<VectorXf, 0, Eigen::Stride<1, 3>>(&wts(2), weights.size()) = weights;
80
81 const auto rows = shapeVectors_.rows();
82 const auto cols = shapeVectors_.cols();
83 MatrixXf A(rows + cols, cols);
84 for (int i = 0; i < wts.size(); i++) {
85 A.row(i) = shapeVectors_.row(i) * wts(i);
86 }
87 A.block(rows, 0, cols, cols) = MatrixXf::Identity(cols, cols) * regularization;
88
89 VectorXf b = VectorXf::Zero(rows + cols);
90 b.topRows(rows) = (vertexVec - baseVec).cwiseProduct(wts);
91
92 return (A.transpose() * A).ldlt().solve(A.transpose() * b);
93 }
94}

115bool BlendShape::isApprox(const BlendShape& blendShape) const {
116 if (baseShape_.size() != blendShape.getBaseShape().size()) {
117 return false;
118 }
119 for (size_t i = 0; i < baseShape_.size(); i++) {
120 if (!baseShape_[i].isApprox(blendShape.getBaseShape()[i])) {
121 return false;
122 }
123 }
124
125 return shapeVectors_.isApprox(blendShape.getShapeVectors()) &&
126 (factorizationValid_ == blendShape.getFactorizationValid());
127}
