68template <typename T>
69void CharacterStateT<T>::set(
70 const CharacterParameters& inParameters,
71 const Character& referenceCharacter,
72 bool updateMesh,
73 bool updateCollision,
74 bool applyLimits) {
75 parameters = inParameters;
76 if (parameters.offsets.size() == 0) {
77 parameters.offsets = VectorXf::Zero(referenceCharacter.parameterTransform.numJointParameters());
78 }
79 MT_CHECK(
80 parameters.pose.size() == referenceCharacter.parameterTransform.numAllModelParameters(),
81 "{} is not {}",
82 parameters.pose.size(),
83 referenceCharacter.parameterTransform.numAllModelParameters());
84 MT_CHECK(
85 parameters.offsets.size() == referenceCharacter.parameterTransform.numJointParameters(),
86 "{} is not {}",
87 parameters.offsets.size(),
88 referenceCharacter.parameterTransform.numJointParameters());
89
90 // Skeleton state must be computed first — locator, mesh, and collision updates below
91 // all read from it.
92 JointParameters jointParams = referenceCharacter.parameterTransform.apply(parameters);
93 if (applyLimits) {
94 jointParams = applyPassiveJointParameterLimits(referenceCharacter.parameterLimits, jointParams);
95 }
96 skeletonState.set(jointParams, referenceCharacter.skeleton);
97
98 locatorState.update(skeletonState, referenceCharacter.locators);
99
100 if (updateMesh && referenceCharacter.mesh && referenceCharacter.skinWeights) {
101 // (Re)allocate the cached mesh whenever vertex count diverges from the reference
102 // — covers both first call and switching to a different reference character.
103 if (!meshState || meshState->vertices.size() != referenceCharacter.mesh->vertices.size()) {
104 meshState = std::make_unique<Mesh>(*referenceCharacter.mesh);
105 }
106
107 // Pose blendshapes take precedence: corrective vertex offsets driven by joint pose are
108 // applied before linear skinning so the SSD pass deforms the corrected bind-pose vertices.
109 if (referenceCharacter.poseShapes) {
110 const auto vs = referenceCharacter.poseShapes->compute(skeletonState);
111
112 meshState->vertices = applySSD(
113 referenceCharacter.inverseBindPose, *referenceCharacter.skinWeights, vs, skeletonState);
114
115 meshState->updateNormals();
116 } else if (referenceCharacter.blendShape) {
117 // `Character` is now non-templated (T270138855), so this branch no longer needs the
118 // float-vs-double specialization workaround.
119 skinWithBlendShapes(referenceCharacter, skeletonState, parameters.pose, *meshState);
120 } else {
121 applySSD(
122 referenceCharacter.inverseBindPose,
123 *referenceCharacter.skinWeights,
124 *referenceCharacter.mesh,
125 skeletonState,
126 *meshState);
127 }
128 }
129
130 if (updateCollision && referenceCharacter.collision) {
131 if (!collisionState || collisionState->origin.size() != referenceCharacter.collision->size()) {
132 collisionState = std::make_unique<CollisionGeometryState>();
133 }
134 collisionState->update(skeletonState, *referenceCharacter.collision);
135 }
136}
