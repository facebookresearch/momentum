21template <typename T>
22void JointStateT<T>::set(
23 const JointT<T>& joint,
24 const JointVectorT<T>& parameters,
25 const JointStateT<T>* parentState,
26 bool computeDeriv) noexcept {
27 derivDirty = !computeDeriv;
28
29 TransformT<T> parent;
30 if (parentState != nullptr) {
31 parent = parentState->transform;
32 }
33
34 // Translation acts in the parent's frame, so the per-axis translation derivative is
35 // the parent's rotated unit axis. Identity when there is no parent.
36 if (computeDeriv) {
37 if (parentState != nullptr) {
38 translationAxis = parentState->transform.toLinear();
39 } else {
40 translationAxis.setIdentity();
41 }
42 }
43
44 localTransform.translation.noalias() = joint.translationOffset + parameters.template head<3>();
45
46 localTransform.rotation = joint.preRotation;
47
48 // Apply rotations in reverse order (Z, Y, X) so the resulting product is Rpre * Rx * Ry * Rz.
49 // The rotation derivative axis for index i is the parent rotation composed with the
50 // partially-accumulated local rotation up to (but not including) axis i.
51 for (int index = 2; index >= 0; --index) {
52 if (computeDeriv) {
53 rotationAxis.col(index).noalias() =
54 (parent.rotation * localTransform.rotation) * RotationAxis<T>[index];
55 }
56 localTransform.rotation *=
57 Quaternion<T>(AngleAxis<T>((T)parameters[3 + index], RotationAxis<T>[index]));
58 }
59
60 // Scale parameter is stored as a log2 value: scale = exp2(parameters[6]).
61 // Keeps scale strictly positive and makes the parameter space uniform in log-ratio.
62 localTransform.scale = std::exp2((T)parameters[6]);
63
64 transform = parent * localTransform;
65}

67template <typename T>
68Vector3<T> JointStateT<T>::getRotationDerivative(const size_t index, const Vector3<T>& ref) const {
69 MT_CHECK(!derivDirty, "Derivatives haven't been computed yet.");
