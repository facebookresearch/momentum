31Character addGloveBones(
32 Character character,
33 const GloveConfig& cfg,
34 const std::array<GloveOffset, 2>& offsets,
35 const std::string& prefix) {
36 MT_THROW_IF(prefix.empty(), "addGloveBones: prefix must not be empty.");
37 auto& newSkel = character.skeleton;
38 auto& newTransform = character.parameterTransform;
39 auto& newInvBindPose = character.inverseBindPose;
40
41 for (size_t hand = 0; hand < 2; ++hand) {
42 const auto& wristName = cfg.wristJointNames[hand];
43 MT_THROW_IF(
44 wristName.empty(), "addGloveBones: wrist joint name for hand {} must not be empty.", hand);
45
46 const size_t wristIdx = newSkel.getJointIdByName(wristName);
47 if (wristIdx == kInvalidIndex) {
48 MT_LOGW("addGloveBones: wrist joint '{}' not found, skipping", wristName);
49 continue;
50 }
51
52 // Skip if glove bone already exists
53 const std::string gloveBoneName = prefix + wristName;
54 if (newSkel.getJointIdByName(gloveBoneName) != kInvalidIndex) {
55 continue;
56 }
57
58 // Create a new joint as child of the wrist
59 Joint joint;
60 joint.name = gloveBoneName;
61 joint.parent = wristIdx;
62 joint.translationOffset = offsets[hand].translation;
63 joint.preRotation = eulerToQuaternion<float>(
64 offsets[hand].rotationEulerXYZ.cast<float>(), 0, 1, 2, EulerConvention::Intrinsic);
65
66 newSkel.joints.push_back(joint);
67 newInvBindPose.push_back(Affine3f::Identity());
68
69 // Resize offsets and activeJointParams for the new joint
70 newTransform.offsets.conservativeResize(newTransform.offsets.size() + kParametersPerJoint);
71 newTransform.activeJointParams.conservativeResize(
72 newTransform.activeJointParams.size() + kParametersPerJoint);
73 for (size_t j = newTransform.offsets.size() - kParametersPerJoint;
74 j < static_cast<size_t>(newTransform.offsets.size());
75 ++j) {
76 newTransform.offsets(j) = 0.0f;
77 }
78 for (size_t j = newTransform.activeJointParams.size() - kParametersPerJoint;
79 j < static_cast<size_t>(newTransform.activeJointParams.size());
80 ++j) {
81 newTransform.activeJointParams(j) = true;
82 }
83
84 // Resize the transform matrix rows for the new joint (no new columns)
85 const int newRows = static_cast<int>(newSkel.joints.size() * kParametersPerJoint);
86 const int curCols = static_cast<int>(newTransform.name.size());
87 newTransform.transform.conservativeResize(newRows, curCols);
88 }
89
90 return character;
91}

93Character addGloveCalibrationParameters(
94 Character character,
95 const GloveConfig& cfg,
96 const std::string& prefix) {
97 MT_THROW_IF(prefix.empty(), "addGloveCalibrationParameters: prefix must not be empty.");
98 auto& newSkel = character.skeleton;
99 auto& newTransform = character.parameterTransform;
100
101 ParameterSet gloveSet;
102 std::vector<Eigen::Triplet<float>> triplets;
103
104 // Parameter name suffixes for translation and rotation DOFs
105 static const std::array<std::string, 6> dofNames{"_tx", "_ty", "_tz", "_rx", "_ry", "_rz"};
106 // Corresponding joint parameter indices: TX=0, TY=1, TZ=2, RX=3, RY=4, RZ=5
107 static const std::array<size_t, 6> dofIndices{TX, TY, TZ, RX, RY, RZ};
108
109 for (size_t hand = 0; hand < 2; ++hand) {
110 MT_THROW_IF(
111 cfg.wristJointNames[hand].empty(),
112 "addGloveCalibrationParameters: wrist joint name for hand {} must not be empty.",
113 hand);
114 const std::string gloveBoneName = prefix + cfg.wristJointNames[hand];
115 const size_t jointIdx = newSkel.getJointIdByName(gloveBoneName);
116 if (jointIdx == kInvalidIndex) {
117 MT_LOGW("addGloveCalibrationParameters: glove bone '{}' not found, skipping", gloveBoneName);
118 continue;
119 }
120
121 // Add 6 DOF parameters (TX, TY, TZ, RX, RY, RZ) for this glove bone
122 for (size_t d = 0; d < 6; ++d) {
123 const std::string paramName = gloveBoneName + dofNames[d];
124 const int row = static_cast<int>(jointIdx * kParametersPerJoint + dofIndices[d]);
125 const auto col = static_cast<int>(newTransform.name.size());
126
127 triplets.emplace_back(row, col, 1.0f);
128
129 MT_CHECK(
130 newTransform.name.size() < kMaxModelParams,
131 "Number of model parameters reached the {} limit.",
132 kMaxModelParams);
133 gloveSet.set(newTransform.name.size());
134 newTransform.name.push_back(paramName);
135 }
136 }
137
138 newTransform.parameterSets["gloves"] = gloveSet;
139
140 // Update the parameter transform matrix
141 const int newRows = static_cast<int>(newSkel.joints.size() * kParametersPerJoint);
142 const int newCols = static_cast<int>(newTransform.name.size());
143 newTransform.transform.conservativeResize(newRows, newCols);
144 SparseRowMatrixf additionalTransforms(newRows, newCols);
145 additionalTransforms.setFromTriplets(triplets.begin(), triplets.end());
146 newTransform.transform += additionalTransforms;
147
148 return character;
149}

156std::array<GloveOffset, 2> extractGloveOffsetsFromCharacter(
157 const Character& gloveChar,
158 const CharacterParameters& params,
159 const GloveConfig& cfg) {
160 std::array<GloveOffset, 2> result;
161
162 // Apply parameter transform to get joint parameters
163 const JointParameters jointParams = gloveChar.parameterTransform.apply(params);
164
165 for (size_t hand = 0; hand < 2; ++hand) {
166 const std::string gloveBoneName = "glove_" + cfg.wristJointNames[hand];
167 const size_t jointIdx = gloveChar.skeleton.getJointIdByName(gloveBoneName);
168 if (jointIdx == kInvalidIndex) {
169 continue;
170 }
171
172 const size_t base = jointIdx * kParametersPerJoint;
173 result[hand].translation =
174 Vector3f(jointParams.v(base + TX), jointParams.v(base + TY), jointParams.v(base + TZ));
175 result[hand].rotationEulerXYZ =
176 Vector3f(jointParams.v(base + RX), jointParams.v(base + RY), jointParams.v(base + RZ));
177 }
178
179 return result;
180}

182void bakeGloveOffsetsFromParams(
183 Character& character,
184 const ModelParameters& solvedParams,
185 const Character& solvingCharacter,
186 const std::optional<GloveConfig>& cfg) {
187 if (!cfg) {
188 return;
189 }
190 // Extract offsets from the solving character's calibration parameters
191 CharacterParameters params;
192 params.pose = solvedParams;
193 params.offsets =
194 JointParameters::Zero(solvingCharacter.skeleton.joints.size() * kParametersPerJoint);
195 const auto offsets = extractGloveOffsetsFromCharacter(solvingCharacter, params, *cfg);
196
197 for (size_t hand = 0; hand < 2; ++hand) {
198 MT_LOGI(
199 "Baking glove offset for '{}': translation=({:.4f}, {:.4f}, {:.4f}), "
200 "rotation_xyz=({:.4f}, {:.4f}, {:.4f})",
201 cfg->wristJointNames[hand],
202 offsets[hand].translation.x(),
203 offsets[hand].translation.y(),
204 offsets[hand].translation.z(),
205 offsets[hand].rotationEulerXYZ.x(),
206 offsets[hand].rotationEulerXYZ.y(),
207 offsets[hand].rotationEulerXYZ.z());
208 }
209
210 // Add glove bones with the solved offsets baked into the joint
211 character = addGloveBones(character, *cfg, offsets);
212}

214std::vector<std::vector<JointToJointPositionDataT<float>>> createGlovePositionConstraintData(
215 std::span<const GloveFrameData> gloveData,
216 const Character& character,
217 const GloveConfig& cfg,
218 size_t handIndex) {
219 if (gloveData.empty() || handIndex >= 2) {
220 return {};
221 }
222
223 std::vector<std::vector<JointToJointPositionDataT<float>>> results(gloveData.size());
224
225 // Find the glove bone for this hand; fall back to the wrist joint itself
226 const std::string gloveBoneName = "glove_" + cfg.wristJointNames[handIndex];
227 size_t referenceIdx = character.skeleton.getJointIdByName(gloveBoneName);
228 if (referenceIdx == kInvalidIndex) {
229 referenceIdx = character.skeleton.getJointIdByName(cfg.wristJointNames[handIndex]);
230 if (referenceIdx == kInvalidIndex) {
231 return results;
232 }
233 }
234
235 // Build name-to-index map for the skeleton
236 std::unordered_map<std::string, size_t> jointLookup;
237 jointLookup.reserve(character.skeleton.joints.size());
238 for (size_t i = 0; i < character.skeleton.joints.size(); ++i) {
239 jointLookup[character.skeleton.joints[i].name] = i;
240 }
241
242 for (size_t iFrame = 0; iFrame < results.size(); ++iFrame) {
243 results[iFrame].reserve(gloveData[iFrame].size());
244 for (const auto& obs : gloveData[iFrame]) {
245 if (!obs.valid) {
246 continue;
247 }
248
249 auto it = jointLookup.find(obs.jointName);
250 if (it == jointLookup.end()) {
251 continue;
252 }
253
254 JointToJointPositionDataT<float> constraint;
255 constraint.sourceJoint = it->second;
256 constraint.sourceOffset = Vector3f::Zero();
257 constraint.referenceJoint = referenceIdx;
258 constraint.referenceOffset = Vector3f::Zero();
259 constraint.target = obs.position;
260 constraint.weight = 1.0f;
261 constraint.name = obs.jointName;
262 results[iFrame].push_back(std::move(constraint));
263 }
264 }
265
266 return results;
267}

322void addGloveConstraintsToSequenceSolver(
323 SequenceSolverFunctionT<float>& solverFunc,
324 size_t solverFrame,
325 const Character& character,
326 const std::vector<std::vector<JointToJointPositionDataT<float>>>& posData,
327 const std::vector<std::vector<JointToJointOrientationDataT<float>>>& oriData,
328 size_t iFrame,
329 float posWeight,
330 float oriWeight) {
331 // Add position constraints for this frame
332 if (iFrame < posData.size() && !posData[iFrame].empty()) {
333 auto posFunc = std::make_shared<JointToJointPositionErrorFunctionT<float>>(character);
334 for (const auto& c : posData[iFrame]) {
335 posFunc->addConstraint(c);
336 }
337 posFunc->setWeight(PositionErrorFunction::kLegacyWeight * posWeight);
338 solverFunc.addErrorFunction(solverFrame, posFunc);
339 }
340
341 // Add orientation constraints for this frame
342 if (iFrame < oriData.size() && !oriData[iFrame].empty()) {
343 auto oriFunc = std::make_shared<JointToJointOrientationErrorFunctionT<float>>(character);
344 for (const auto& c : oriData[iFrame]) {
345 oriFunc->addConstraint(c);
346 }
347 oriFunc->setWeight(OrientationErrorFunction::kLegacyWeight * oriWeight);
348 solverFunc.addErrorFunction(solverFrame, oriFunc);
349 }

352std::optional<GloveErrorFunctions> setupGloveErrorFunctions(
353 SkeletonSolverFunctionT<float>& solverFunc,
354 const Character& character,
355 std::span<const GloveFrameData> gloveData,
356 const GloveConfig& cfg,
357 size_t handIndex) {
358 if (gloveData.empty()) {
359 return std::nullopt;
360 }
361
362 // Verify a reference joint exists (glove bone preferred, wrist as fallback)
363 if (handIndex >= 2) {
364 return std::nullopt;
365 }
366 const std::string gloveBoneName = "glove_" + cfg.wristJointNames[handIndex];
367 size_t referenceIdx = character.skeleton.getJointIdByName(gloveBoneName);
368 if (referenceIdx == kInvalidIndex) {
369 referenceIdx = character.skeleton.getJointIdByName(cfg.wristJointNames[handIndex]);
370 if (referenceIdx == kInvalidIndex) {
371 return std::nullopt;
372 }
373 }
374
375 GloveErrorFunctions funcs;
376
377 // Build constraint data
378 funcs.posData = createGlovePositionConstraintData(gloveData, character, cfg, handIndex);
379 funcs.oriData = createGloveOrientationConstraintData(gloveData, character, cfg, handIndex);
380
381 // Create and register error functions
382 funcs.posFunc = std::make_shared<JointToJointPositionErrorFunctionT<float>>(character);
383 funcs.posFunc->setWeight(PositionErrorFunction::kLegacyWeight * cfg.positionWeight);
384 solverFunc.addErrorFunction(funcs.posFunc);
385
386 funcs.oriFunc = std::make_shared<JointToJointOrientationErrorFunctionT<float>>(character);
387 funcs.oriFunc->setWeight(OrientationErrorFunction::kLegacyWeight * cfg.orientationWeight);
388 solverFunc.addErrorFunction(funcs.oriFunc);
389
390 return funcs;
391}

393void updateGloveConstraintsForFrame(GloveErrorFunctions& funcs, size_t iFrame) {
394 // Update position constraints
395 funcs.posFunc->clearConstraints();
396 if (iFrame < funcs.posData.size()) {
397 for (const auto& c : funcs.posData[iFrame]) {
398 funcs.posFunc->addConstraint(c);
399 }
400 }
401
402 // Update orientation constraints
403 funcs.oriFunc->clearConstraints();
404 if (iFrame < funcs.oriData.size()) {
405 for (const auto& c : funcs.oriData[iFrame]) {
406 funcs.oriFunc->addConstraint(c);
407 }
408 }
409}
