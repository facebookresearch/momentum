14std::vector<MarkerTrack> groupMarkerTracks(
15 std::span<const std::vector<Marker>> frames,
16 std::optional<float> minConfidence) {
17 std::vector<MarkerTrack> tracks;
18 std::unordered_map<std::string, size_t> trackByName;
19 for (size_t frame = 0; frame < frames.size(); ++frame) {
20 for (const Marker& marker : frames[frame]) {
21 if (marker.occluded || (minConfidence.has_value() && marker.confidence < *minConfidence)) {
22 continue;
23 }
24 const auto [entry, added] = trackByName.try_emplace(marker.name, tracks.size());
25 if (added) {
26 tracks.emplace_back().name = marker.name;
27 }
28 MarkerTrack& track = tracks.at(entry->second);
29 track.frames.push_back(frame);
30 track.positions.push_back(marker.pos);
31 track.confidences.push_back(marker.confidence);
32 }
33 }
34 return tracks;
35}
