24// Converts Eigen matrix or vector types to nlohmann::json
25//
26// Note: We intentionally avoid defining nlohmann::adl_serializers for type conversion between Eigen
27// types and nlohmann::json to prevent multiple definitions of the same type in different projects
28// that are compiled in a single translation unit. Instead, we recommend using explicit conversion
29// between the types using the toJson() and fromJson() functions provided in this header.
30template <typename Derived>
31void toJson(const Eigen::MatrixBase<Derived>& mat, nlohmann::json& j) {
32 // always save new format that's numpy compatible
33 // this works for both dynamic and statically sized matrices
34 j = nlohmann::json::array();
35 for (int col_i = 0; col_i < mat.cols(); ++col_i) {
36 nlohmann::json jcol = nlohmann::json::array();
37 for (int row_i = 0; row_i < mat.rows(); ++row_i) {
38 jcol.push_back(mat(row_i, col_i));
39 }
40 j.push_back(jcol);
41 }
42}

44// Converts nlohmann::json to Eigen matrix or vector types
45template <typename T>
46T fromJson(const nlohmann::json& j) {
47 T result;
48
49 // partially dynamic matrix
50 if constexpr (T::RowsAtCompileTime == Eigen::Dynamic || T::ColsAtCompileTime == Eigen::Dynamic) {
51 // dynamic size matrix
52 // reads dimensions from the json
53 const auto numCols = j.size();
54 if (numCols > 0) {
55 const auto numRows = j.at(0).size();
56 result.resize(numRows, numCols);
57
58 std::vector<typename T::Scalar> coef;
59 for (int col_i = 0; col_i < numCols; ++col_i) {
60 coef = j[col_i];
61 std::copy(coef.begin(), coef.end(), result.col(col_i).begin());
62 }
63 }
64 }
65 // fixed size matrix or vector
66 else {
67 const int Rows = T::RowsAtCompileTime;
68 const int Cols = T::ColsAtCompileTime;
69
70 // vector stored as array
71 if (Rows == 1 && Cols != -1 && j.size() == Cols &&
72 j[0].type() != nlohmann::json::value_t::array) {
73 std::vector<typename T::Scalar> coef = j;
74 std::copy(coef.begin(), coef.end(), result.row(0).begin());
75 } else if (
76 Cols == 1 && Rows != -1 && j.size() == Rows &&
77 j[0].type() != nlohmann::json::value_t::array) {
78 std::vector<typename T::Scalar> coef = j;
79 std::copy(coef.begin(), coef.end(), result.col(0).begin());
80 }
81 // fixed size matrix
82 else {
83 const auto numCols = j.size();
84 if (numCols > 0) {
85 for (int col_i = 0; col_i < Cols; ++col_i) {
86 std::vector<typename T::Scalar> coef = j[col_i];
87 std::copy(coef.begin(), coef.end(), result.col(col_i).begin());
88 }
89 }
90 }
91 }
92
93 return result;
94}
